# ROS 2 Action Execution Pipeline

This chapter covers executing ROS 2 action sequences generated by the LLM planning system. We'll explore how to map high-level plans to specific ROS 2 actions and handle execution with feedback and error recovery.

## Introduction to ROS 2 Actions

ROS 2 actions are a way to communicate long-running tasks between nodes. Unlike services, actions can provide feedback during execution and can be preempted. This makes them ideal for robot tasks that take time to complete, like navigation or manipulation.

## Mapping LLM Plans to ROS 2 Actions

The VLA system bridges cognitive planning with physical execution through action mapping. Here's the process:

1. LLM generates a plan with high-level actions
2. Action mapper converts to specific ROS 2 action calls
3. ROS 2 executes the actions on the robot
4. Feedback is collected and sent back to the planning system

```python
class ActionMapper:
    def __init__(self):
        # Initialize ROS 2 clients for various action types
        self.navigation_client = NavigationActionClient()
        self.manipulation_client = ManipulationActionClient()
        self.vision_client = VisionActionClient()

    def execute_plan(self, llm_plan):
        """Execute an LLM-generated plan as ROS 2 actions"""
        results = []

        for task in llm_plan.sub_tasks:
            action_type = task["action"]
            parameters = task["parameters"]

            try:
                # Map the action to appropriate ROS 2 client
                result = self._execute_action(action_type, parameters)
                results.append({
                    "task": task,
                    "result": result,
                    "status": "success"
                })
            except Exception as e:
                results.append({
                    "task": task,
                    "error": str(e),
                    "status": "failed"
                })

                # Handle error - maybe retry or skip
                if not self._handle_error(task, e):
                    break  # Stop execution if critical error

        return results

    def _execute_action(self, action_type, parameters):
        """Execute a specific action based on its type"""
        if action_type == "navigate_to":
            return self.navigation_client.send_goal(
                target_pose=parameters["target"],
                speed=parameters.get("speed", 0.5)
            )
        elif action_type == "grasp_object":
            return self.manipulation_client.send_grasp_goal(
                object_id=parameters["object_id"],
                grasp_type=parameters.get("grasp_type", "default")
            )
        elif action_type == "place_object":
            return self.manipulation_client.send_place_goal(
                object_id=parameters["object_id"],
                location=parameters["location"]
            )
        elif action_type == "locate_object":
            return self.vision_client.send_detection_goal(
                object_type=parameters["object_type"],
                search_area=parameters.get("search_area", "current_view")
            )
        else:
            raise ValueError(f"Unknown action type: {action_type}")
```

## Action Execution with Feedback

For reliable execution, we need to monitor action progress and handle feedback:

```python
import rclpy
from rclpy.action import ActionClient
from rclpy.node import Node

class FeedbackActionExecutor(Node):
    def __init__(self):
        super().__init__('feedback_action_executor')

        # Action clients for different types of actions
        self.nav_client = ActionClient(self, NavigateToPose, 'navigate_to_pose')
        self.manip_client = ActionClient(self, FollowJointTrajectory, 'manipulator_controller/follow_joint_trajectory')

    def execute_with_feedback(self, action_type, goal, timeout=60.0):
        """Execute an action with feedback monitoring"""
        if action_type == "navigation":
            return self._execute_navigation_with_feedback(goal, timeout)
        elif action_type == "manipulation":
            return self._execute_manipulation_with_feedback(goal, timeout)
        else:
            raise ValueError(f"Unsupported action type: {action_type}")

    def _execute_navigation_with_feedback(self, goal, timeout):
        """Execute navigation with continuous feedback"""
        self.nav_client.wait_for_server()

        # Send the goal
        future = self.nav_client.send_goal_async(
            goal,
            feedback_callback=self._navigation_feedback_callback
        )

        # Wait for result with timeout
        rclpy.spin_until_future_complete(self, future, timeout_sec=timeout)

        if future.result() is not None:
            goal_handle = future.result()
            if goal_handle.accepted:
                result_future = goal_handle.get_result_async()
                rclpy.spin_until_future_complete(self, result_future, timeout_sec=timeout)

                if result_future.result() is not None:
                    return result_future.result().result
                else:
                    raise RuntimeError("Navigation result future not completed")
            else:
                raise RuntimeError("Navigation goal was rejected")
        else:
            raise RuntimeError("Navigation goal was not accepted")

    def _navigation_feedback_callback(self, feedback_msg):
        """Handle navigation feedback"""
        self.get_logger().info(f'Navigation progress: {feedback_msg.feedback.distance_remaining:.2f}m remaining')

        # Update progress in VLA system
        self._update_vla_progress({
            "action_type": "navigation",
            "progress": feedback_msg.feedback.distance_remaining,
            "status": "in_progress"

    def _execute_manipulation_with_feedback(self, goal, timeout):
        """Execute manipulation with continuous feedback"""
        self.manip_client.wait_for_server()

            goal,

        rclpy.spin_until_future_complete(self, future, timeout_sec=timeout)

        if future.result() is not None:
            goal_handle = future.result()
            if goal_handle.accepted:
                result_future = goal_handle.get_result_async()
                rclpy.spin_until_future_complete(self, result_future, timeout_sec=timeout)

                if result_future.result() is not None:
                    return result_future.result().result
                else:
                    raise RuntimeError("Manipulation result future not completed")
            else:
                raise RuntimeError("Manipulation goal was rejected")
        else:
            raise RuntimeError("Manipulation goal was not accepted")
```

## Error Handling and Recovery

Robust action execution requires comprehensive error handling:

```python
class RobustActionExecutor:
    def __init__(self):
        self.max_retries = 3
        self.retry_delay = 2.0

    async def execute_action_with_recovery(self, action_type, parameters):
        """Execute an action with error handling and recovery"""
        for attempt in range(self.max_retries):
            try:
                result = await self._execute_single_action(action_type, parameters)

                if self._is_successful(result):
                    return result
                else:
                    self.get_logger().warning(f"Action failed on attempt {attempt + 1}, result: {result}")

            except Exception as e:
                self.get_logger().error(f"Action execution failed on attempt {attempt + 1}: {e}")

                if attempt == self.max_retries - 1:
                    # Last attempt, try alternative approach
                    alternative_result = await self._try_alternative_action(action_type, parameters)
                    if alternative_result:
                        return alternative_result
                    else:
                        raise e

            # Wait before retry
            if attempt < self.max_retries - 1:
                await asyncio.sleep(self.retry_delay * (2 ** attempt))  # Exponential backoff

    def _is_successful(self, result):
        """Determine if an action result is successful"""
        if hasattr(result, 'success'):
            return result.success
        elif hasattr(result, 'status'):
            return result.status == 'SUCCESS'
        else:
            # Default to considering it successful if no explicit failure
            return True

    async def _try_alternative_action(self, action_type, parameters):
        """Try an alternative approach if primary action fails"""
        # For navigation, try a different path
        if action_type == "navigate_to":
            alternative_params = parameters.copy()
            alternative_params['planning_strategy'] = 'alternative_path'
            return await self._execute_single_action(action_type, alternative_params)

        # For manipulation, try a different grasp
        elif action_type == "grasp_object":
            alternative_params = parameters.copy()
            alternative_params['grasp_type'] = 'alternative_grasp'
            return await self._execute_single_action(action_type, alternative_params)

        # For other actions, no alternative available
        return None
```

## Action Sequences and Coordination

For complex tasks, multiple actions need to be coordinated:

```python
class ActionSequenceExecutor:
    def __init__(self):
        self.executor = RobustActionExecutor()
        self.current_state = {}  # Track robot state between actions

    async def execute_sequence(self, action_sequence):
        """Execute a sequence of actions with state tracking"""
        results = []

        for i, action in enumerate(action_sequence):
            try:
                # Update action parameters based on current state
                updated_params = self._update_parameters_with_state(action['parameters'], self.current_state)

                # Execute the action
                result = await self.executor.execute_action_with_recovery(
                    action['action'],
                    updated_params
                )

                # Update state based on action result
                self._update_state(action['action'], result)

                results.append({
                    "action_index": i,
                    "action": action,
                    "result": result,
                    "timestamp": time.time()
                })

            except Exception as e:
                # Log error and decide whether to continue
                error_info = {
                    "action_index": i,
                    "action": action,
                    "error": str(e),
                    "timestamp": time.time()
                }

                if self._is_critical_error(e):
                    # Critical error - stop execution
                    results.append(error_info)
                    break
                else:
                    # Non-critical error - continue with next action
                    results.append(error_info)
                    continue

        return results

    def _update_parameters_with_state(self, parameters, state):
        """Update action parameters based on current robot state"""
        updated = parameters.copy()

        # If target location is relative to current position
        if 'relative_to' in updated:
            if updated['relative_to'] == 'current_position':
                updated['target'] = self._compute_absolute_position(
                    state.get('current_position'),
                    updated['offset']
                )

        # If object location needs to be updated based on perception
        if 'object_id' in updated and updated['object_id'] in state.get('object_locations', {}):
            updated['target'] = state['object_locations'][updated['object_id']]

        return updated

    def _update_state(self, action_type, result):
        """Update the robot state based on action result"""
        if action_type == "navigate_to" and 'result_location' in result:
            self.current_state['current_position'] = result['result_location']

        elif action_type == "grasp_object" and result.get('success'):
            # Remove object from environment since it's now held
            obj_id = result.get('object_id')
            if 'object_locations' in self.current_state and obj_id:
                self.current_state['object_locations'].pop(obj_id, None)

        elif action_type == "place_object" and result.get('success'):
            # Add object to new location
            obj_id = result.get('object_id')
            new_location = result.get('location')
            if obj_id and new_location:
                if 'object_locations' not in self.current_state:
                    self.current_state['object_locations'] = {}
                self.current_state['object_locations'][obj_id] = new_location
```

## Integration with VLA System

The action execution system integrates with the broader VLA pipeline:

```python
class VLAActionExecutor:
    def __init__(self):
        self.action_executor = ActionSequenceExecutor()
        self.progress_publisher = None  # ROS 2 publisher for progress updates
        self.feedback_handler = None    # For handling feedback from VLA system

    async def execute_vla_plan(self, llm_plan):
        """Execute an LLM plan through the VLA action system"""

        # Convert LLM plan to ROS 2 action sequence
        action_sequence = self._llm_plan_to_action_sequence(llm_plan)

        # Execute the sequence
        results = await self.action_executor.execute_sequence(action_sequence)

        # Process results and update VLA system
        final_result = self._process_execution_results(results, llm_plan)

        # Send completion feedback to VLA system
        await self._send_completion_feedback(final_result)

        return final_result

    def _llm_plan_to_action_sequence(self, llm_plan):
        """Convert LLM plan to ROS 2 action sequence"""
        sequence = []

        for task in llm_plan.sub_tasks:
            # Map LLM action to ROS 2 action
            ros2_action = self._map_action(task)
            sequence.append(ros2_action)

        return sequence

    def _map_action(self, task):
        """Map an LLM task to a ROS 2 action"""
        action_type = task["action"]
        parameters = task["parameters"]

        # Define mapping from LLM actions to ROS 2 actions
        action_mapping = {
            "navigate_to": {
                "action": "navigate_to_pose",
                "parameters": {
                    "pose": parameters.get("target"),
                    "speed": parameters.get("speed", 0.5)
                }
            },
            "grasp_object": {
                "action": "grasp",
                "parameters": {
                    "object_id": parameters.get("object_id"),
                    "grasp_type": parameters.get("grasp_type", "top_grasp")
                }
            },
            "place_object": {
                "action": "place",
                "parameters": {
                    "object_id": parameters.get("object_id"),
                    "location": parameters.get("location")
                }
            },
            "locate_object": {
                "action": "detect_objects",
                "parameters": {
                    "object_type": parameters.get("object_type"),
                    "search_area": parameters.get("search_area", "current_view")
                }
            }
        }

        if action_type in action_mapping:
            return action_mapping[action_type]
        else:
            raise ValueError(f"Cannot map LLM action '{action_type}' to ROS 2 action")
```
