# Mini-Project Code Example: Voice-Controlled Task Runner

This chapter provides the complete, runnable code for the voice-controlled task runner mini-project that integrates all VLA components.

## Complete Implementation

### 1. Main Application Entry Point

```python
# main.py
import asyncio
import logging
from fastapi import FastAPI, WebSocket, WebSocketDisconnect
from pydantic import BaseModel
from typing import Dict, Any, Optional
import base64
from vla_system import VoiceControlledTaskRunner

# Set up logging
logging.basicConfig(level=logging.INFO)
logger = logging.getLogger(__name__)

# Initialize FastAPI app
app = FastAPI(title="VLA Voice-Controlled Task Runner", version="1.0.0")

# Global task runner instance
task_runner = None

@app.on_event("startup")
async def startup_event():
    """Initialize the task runner on startup"""
    global task_runner
    task_runner = VoiceControlledTaskRunner()
    logger.info("Voice-controlled task runner initialized")

@app.get("/")
async def root():
    """Root endpoint for health check"""
    return {"message": "VLA Voice-Controlled Task Runner API", "status": "running"}

class VoiceTaskRequest(BaseModel):
    """Request model for voice task execution"""
    audio_data: str  # Base64 encoded audio data
    camera_image_data: Optional[str] = None  # Optional camera image for perception
    task_context: Optional[Dict[str, Any]] = {}

@app.post("/voice-task")
async def execute_voice_task(request: VoiceTaskRequest):
    """Execute a voice-controlled task"""
    global task_runner

    try:
        result = await task_runner.process_voice_task(
            audio_data=request.audio_data,
            camera_image_data=request.camera_image_data,
            task_context=request.task_context
        )

        return result
    except Exception as e:
        logger.error(f"Error executing voice task: {str(e)}")
        return {
            "success": False,
            "error": str(e),
            "summary": f"Task failed: {str(e)}"
        }

# WebSocket endpoint for real-time interaction
@app.websocket("/ws/task-status")
async def websocket_task_status(websocket: WebSocket):
    """WebSocket endpoint for real-time task status updates"""
    await websocket.accept()

    try:
        while True:
            # Listen for messages from client
            data = await websocket.receive_text()

            # Process the command (in this example, we just echo back)
            response = {"message": f"Received: {data}", "status": "acknowledged"}
            await websocket.send_text(str(response))

    except WebSocketDisconnect:
        logger.info("WebSocket disconnected")
```

### 2. Core Voice Control System

```python
# vla_system.py
import asyncio
import base64
import logging
from typing import Dict, Any, Optional
from datetime import datetime

# Import services (these would be the actual service classes we've created)
from backend.src.services.whisper_service import WhisperService
from backend.src.services.llm_planning_service import LLMPlanningService
from backend.src.services.vision_service import VisionService
from backend.src.services.ros2_integration import ROS2IntegrationService
from backend.src.services.ros2_mapping_service import ROS2MappingService
from backend.src.services.perception_planning_integration import PerceptionPlanningIntegration

class VoiceControlledTaskRunner:
    """
    Complete voice-controlled task runner system
    """
    def __init__(self):
        # Initialize all VLA services
        self.whisper_service = WhisperService()
        self.planning_service = LLMPlanningService()
        self.vision_service = VisionService()
        self.ros2_mapping_service = ROS2MappingService()
        self.ros2_integration_service = ROS2IntegrationService()
        self.perception_planning_integration = PerceptionPlanningIntegration()

        # Connect to ROS 2 system
        asyncio.create_task(self._initialize_ros2())

    async def _initialize_ros2(self):
        """Initialize connection to ROS 2 system"""
        try:
            connected = await self.ros2_integration_service.connect_to_ros2()
            if connected:
                logging.info("Successfully connected to ROS 2 system")
            else:
                logging.warning("Could not connect to ROS 2 system - running in simulation mode")
        except Exception as e:
            logging.error(f"Error initializing ROS 2 connection: {e}")

    async def process_voice_task(
        self,
        audio_data: str,
        camera_image_data: Optional[str] = None,
        task_context: Optional[Dict[str, Any]] = None
    ) -> Dict[str, Any]:
        """
        Process a complete voice-controlled task
        """
        start_time = datetime.utcnow()
        logger = logging.getLogger(__name__)
        logger.info("Starting voice-controlled task processing")

        try:
            # Phase 1: Voice Processing
            logger.info("Phase 1: Processing voice command")
            voice_result = await self._process_voice_command(audio_data)

            if not voice_result["success"]:
                return {
                    "success": False,
                    "error": voice_result["error"],
                    "phase": "voice_processing",
                    "start_time": start_time.isoformat(),
                    "end_time": datetime.utcnow().isoformat(),
                    "duration": (datetime.utcnow() - start_time).total_seconds(),
                    "summary": "Failed to process voice command"
                }

            command_text = voice_result["command"]
            logger.info(f"Recognized command: '{command_text}'")

            # Phase 2: Perception Integration (if available)
            perception_context = {}
            if camera_image_data:
                logger.info("Phase 2: Integrating perception data")
                perception_result = await self._integrate_perception(camera_image_data, command_text)
                perception_context = perception_result.get("enhanced_context", {})
                logger.info(f"Incorporated perception data with {len(perception_context.get('perception', {}).get('detected_objects', []))} objects")

            # Phase 3: LLM Planning
            logger.info("Phase 3: Generating LLM plan")
            plan = await self._generate_plan(command_text, perception_context)
            logger.info(f"Generated plan with {len(plan.sub_tasks)} sub-tasks")

            # Phase 4: Action Mapping
            logger.info("Phase 4: Mapping plan to ROS 2 actions")
            action_sequence = await self._map_to_ros2_actions(plan)
            logger.info(f"Mapped to {len(action_sequence.actions)} ROS 2 actions")

            # Phase 5: Execution
            logger.info("Phase 5: Executing action sequence")
            execution_result = await self._execute_action_sequence(action_sequence)

            # Phase 6: Result Compilation
            result = {
                "success": True,
                "command": command_text,
                "transcription_confidence": voice_result["confidence"],
                "plan_tasks": len(plan.sub_tasks),
                "actions_executed": len(action_sequence.actions),
                "execution_status": execution_result["status"],
                "execution_result": execution_result,
                "perception_used": bool(camera_image_data),
                "start_time": start_time.isoformat(),
                "end_time": datetime.utcnow().isoformat(),
                "duration": (datetime.utcnow() - start_time).total_seconds(),
                "summary": f"Successfully executed '{command_text}' with {len(plan.sub_tasks)} tasks"
            }

            logger.info(f"Task completed successfully: {result['summary']}")
            return result

        except Exception as e:
            logger.error(f"Error in voice-controlled task processing: {str(e)}")
            return {
                "success": False,
                "error": str(e),
                "start_time": start_time.isoformat(),
                "end_time": datetime.utcnow().isoformat(),
                "duration": (datetime.utcnow() - start_time).total_seconds(),
                "summary": f"Task failed due to error: {str(e)}"
            }

    async def _process_voice_command(self, audio_data: str) -> Dict[str, Any]:
        """Process voice command using Whisper"""
        try:
            transcription_result = await self.whisper_service.transcribe_audio(audio_data)

            if transcription_result["confidence"] < 0.5:
                return {
                    "success": False,
                    "error": f"Low transcription confidence: {transcription_result['confidence']:.2f}",
                    "confidence": transcription_result["confidence"]
                }

            return {
                "success": True,
                "command": transcription_result["text"],
                "confidence": transcription_result["confidence"]
            }
        except Exception as e:
            return {
                "success": False,
                "error": f"Voice processing error: {str(e)}",
                "confidence": 0.0
            }

    async def _integrate_perception(self, image_data: str, command_text: str) -> Dict[str, Any]:
        """Integrate perception data with the command"""
        try:
            # Create perception-aware context
            enhanced_context = await self.perception_planning_integration.create_perception_enhanced_context(
                user_query=command_text,
                image_data=image_data
            )

            return {
                "success": True,
                "enhanced_context": enhanced_context
            }
        except Exception as e:
            logging.warning(f"Perception integration failed: {str(e)}, proceeding without perception")
            return {
                "success": True,
                "enhanced_context": {}
            }

    async def _generate_plan(self, command_text: str, context: Dict[str, Any]):
        """Generate LLM plan for the command"""
        try:
            # Create plan with or without context
            plan = await self.planning_service.create_plan_from_goal(
                goal=command_text,
                context=context
            )

            # Validate the plan
            is_valid, issues = await self.planning_service.validate_plan(plan)

            if not is_valid:
                logging.warning(f"Plan validation issues: {issues}")

            return plan
        except Exception as e:
            raise Exception(f"Plan generation failed: {str(e)}")

    async def _map_to_ros2_actions(self, plan) -> Any:
        """Map LLM plan to ROS 2 action sequence"""
        try:
            action_sequence = await self.ros2_mapping_service.map_plan_to_action_sequence(plan)

            # Validate the action sequence
            is_valid, issues = await self.ros2_mapping_service.validate_action_sequence(action_sequence)

            if not is_valid:
                logging.warning(f"Action sequence validation issues: {issues}")

            return action_sequence
        except Exception as e:
            raise Exception(f"Action mapping failed: {str(e)}")

    async def _execute_action_sequence(self, action_sequence) -> Dict[str, Any]:
        """Execute the ROS 2 action sequence"""
        try:
            execution_result = await self.ros2_integration_service.execute_action_sequence(action_sequence)
            return execution_result
        except Exception as e:
            raise Exception(f"Action execution failed: {str(e)}")

    async def run_demo_tasks(self) -> Dict[str, Any]:
        """Run a series of demo tasks to test the system"""
        demo_tasks = [
            "Move forward 1 meter",
            "Turn left 90 degrees",
            "Find the red cup",
            "Go to the table"
        ]

        results = []

        for i, task in enumerate(demo_tasks):
            logging.info(f"Running demo task {i+1}/{len(demo_tasks)}: {task}")

            # For demo, we'll use simulated audio data
            simulated_audio = self._create_simulated_audio(task)

            result = await self.process_voice_task(simulated_audio)
            results.append({
                "task": task,
                "result": result
            })

            logging.info(f"Demo task {i+1} completed with success: {result['success']}")

        summary = {
            "total_tasks": len(demo_tasks),
            "successful_tasks": sum(1 for r in results if r["result"]["success"]),
            "failed_tasks": sum(1 for r in results if not r["result"]["success"]),
            "individual_results": results
        }

        return summary

    def _create_simulated_audio(self, text: str) -> str:
        """Create simulated audio data for demo purposes"""
        # This is a placeholder - in real system, you'd capture from microphone
        # For simulation, we'll return a base64 encoded placeholder
        simulated_audio_data = "UklGRigAAABXQVZFZm10IBAAAAABAAEARKwAAIhYAQACABAAZGF0YQQAAAAAAAAAAAA="
        return f"data:audio/wav;base64,{simulated_audio_data}"
```

```html
        class VLATaskRunnerClient {
            constructor() {
                this.recording = false;
                this.mediaRecorder = null;
                this.audioChunks = [];

                this.setupEventListeners();
            }

            setupEventListeners() {
                document.getElementById('startBtn').addEventListener('click', () => this.startRecording());
                document.getElementById('stopBtn').addEventListener('click', () => this.stopRecording());
                document.getElementById('sendBtn').addEventListener('click', () => this.sendCommand());

            async startRecording() {
                try {
                    const stream = await navigator.mediaDevices.getUserMedia({ audio: true });

                    this.mediaRecorder = new MediaRecorder(stream);
                    this.audioChunks = [];

                    this.mediaRecorder.ondataavailable = event => {
                        this.audioChunks.push(event.data);
                    };

                    this.mediaRecorder.onstop = async () => {
                        const audioBlob = new Blob(this.audioChunks, { type: 'audio/wav' });
                        const audioBase64 = await this.blobToBase64(audioBlob);

                        await this.sendVoiceCommand(audioBase64);

                        // Stop all tracks to release microphone
                        stream.getTracks().forEach(track => track.stop());
                    };

                    this.mediaRecorder.start();
                    this.recording = true;

                    document.getElementById('startBtn').disabled = true;
                    document.getElementById('stopBtn').disabled = false;
                    document.getElementById('recordingStatus').style.display = 'block';

                } catch (error) {
                    console.error('Error accessing microphone:', error);
                    alert('Could not access microphone. Please check permissions.');
                }

                if (this.mediaRecorder && this.recording) {
                    this.mediaRecorder.stop();
                    this.recording = false;

                    document.getElementById('startBtn').disabled = false;
                    document.getElementById('stopBtn').disabled = true;
                    document.getElementById('recordingStatus').style.display = 'none';

                            audio_data: audioBase64,
                            task_context: { source: 'web_client' }
                        })
                    });

                    const result = await response.json();
                    this.displayResult(result);

                } catch (error) {
                    console.error('Error sending voice command:', error);
                    statusDiv.innerHTML = `<div class="status error">Error: ${error.message}</div>`;
                }
            }

            async sendCommand() {

                    alert('Please enter a command');
                    return;
                }

                const statusDiv = document.getElementById('taskStatus');
                statusDiv.innerHTML = '<div class="status info">Processing command...</div>';

                try {
                    // For text commands, we'll simulate audio data
                    // In a real system, you might convert text to speech
                    const response = await fetch('/voice-task', {
                        method: 'POST',
                        headers: {
                            'Content-Type': 'application/json',
                        },
                        body: JSON.stringify({
```

### 7. Deployment Scripts

```bash
#!/bin/bash
# deploy.sh

set -e

echo "Starting VLA Voice-Controlled Task Runner deployment..."

# Build the Docker images
echo "Building Docker images..."
docker-compose build

# Start the services
echo "Starting services..."
docker-compose up -d

# Wait for services to be ready
echo "Waiting for services to be ready..."
sleep 10

# Run health check
echo "Running health check..."
HEALTH_STATUS=$(curl -s -o /dev/null -w "%{http_code}" http://localhost:8000/health)

if [ $HEALTH_STATUS -eq 200 ]; then
    echo "✅ Deployment successful! Service is running."
    echo "API available at: http://localhost:8000"
    echo "Health check: http://localhost:8000/health"
else
    echo "❌ Deployment failed. Service returned status: $HEALTH_STATUS"
    docker-compose logs
    exit 1
fi

echo "Deployment completed successfully!"
```
