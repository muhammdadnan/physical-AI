# ROS 2 Mini-Projects

## Project Overview

In this section, we'll implement two hands-on mini-projects that demonstrate practical application of ROS 2 concepts. These projects will solidify your understanding of publisher-subscriber systems and service-based robot control.

## Mini-Project 1: Sensor Publisher-Subscriber System

### Objective
Create a publisher that simulates sensor data (IMU and Depth) and a subscriber that processes this data.

### Implementation Steps

#### 1. Create the Publisher Node

```python
import rclpy
from rclpy.node import Node
from sensor_msgs.msg import Imu, Range
import random

class SensorPublisher(Node):
    def __init__(self):
        super().__init__('sensor_publisher')
        self.imu_publisher = self.create_publisher(Imu, 'imu_data', 10)
        self.depth_publisher = self.create_publisher(Range, 'depth_data', 10)
        timer_period = 0.1  # seconds
        self.timer = self.create_timer(timer_period, self.timer_callback)

    def timer_callback(self):
        # Publish IMU data
        imu_msg = Imu()
        imu_msg.header.stamp = self.get_clock().now().to_msg()
        imu_msg.header.frame_id = 'imu_link'

        # Simulate IMU values
        imu_msg.linear_acceleration.x = random.uniform(-9.8, 9.8)
        imu_msg.linear_acceleration.y = random.uniform(-9.8, 9.8)
        imu_msg.linear_acceleration.z = random.uniform(-9.8, 9.8)

        self.imu_publisher.publish(imu_msg)

        # Publish Depth data
        depth_msg = Range()
        depth_msg.header.stamp = self.get_clock().now().to_msg()
        depth_msg.header.frame_id = 'depth_sensor'
        depth_msg.radiation_type = Range.ULTRASOUND
        depth_msg.range = random.uniform(0.1, 3.0)  # meters

        self.depth_publisher.publish(depth_msg)

        self.get_logger().info(f'Published sensor data: IMU accel=({imu_msg.linear_acceleration.x:.2f}, {imu_msg.linear_acceleration.y:.2f}, {imu_msg.linear_acceleration.z:.2f}), Depth={depth_msg.range:.2f}m')

def main(args=None):
    rclpy.init(args=args)
    sensor_publisher = SensorPublisher()

    try:
        rclpy.spin(sensor_publisher)
    except KeyboardInterrupt:
        pass
    finally:
        sensor_publisher.destroy_node()
        rclpy.shutdown()

if __name__ == '__main__':
    main()
```

#### 2. Create the Subscriber Node

```python
import rclpy
from rclpy.node import Node
from sensor_msgs.msg import Imu, Range

class SensorSubscriber(Node):
    def __init__(self):
        super().__init__('sensor_subscriber')
        self.imu_subscription = self.create_subscription(
            Imu,
            'imu_data',
            self.imu_callback,
            10)
        self.depth_subscription = self.create_subscription(
            Range,
            'depth_data',
            self.depth_callback,
            10)

        self.imu_subscription  # prevent unused variable warning
        self.depth_subscription  # prevent unused variable warning

    def imu_callback(self, msg):
        self.get_logger().info(f'IMU Data received: Accel=({msg.linear_acceleration.x:.2f}, {msg.linear_acceleration.y:.2f}, {msg.linear_acceleration.z:.2f})')

    def depth_callback(self, msg):
        self.get_logger().info(f'Depth Data received: Range={msg.range:.2f}m')

        # Simple obstacle detection
        if msg.range < 0.5:
            self.get_logger().warn('Obstacle detected! Distance: {:.2f}m'.format(msg.range))

def main(args=None):
    rclpy.init(args=args)
    sensor_subscriber = SensorSubscriber()

    try:
        rclpy.spin(sensor_subscriber)
    except KeyboardInterrupt:
        pass
    finally:
        sensor_subscriber.destroy_node()
        rclpy.shutdown()

if __name__ == '__main__':
    main()
```

## Mini-Project 2: Joint Control Service

### Objective
Create a service that allows a Python AI agent to control robot joints.

#### 1. Define the Service Interface

Create a service definition file `JointControl.srv`:

```
# Joint name
string joint_name
# Target position
float64 target_position
---
# Success flag
bool success
# Message
string message
```

#### 2. Create the Service Server

```python
import rclpy
from rclpy.node import Node
from your_package.srv import JointControl  # Assuming you have the service definition
import time

class JointControlService(Node):
    def __init__(self):
        super().__init__('joint_control_service')
        self.srv = self.create_service(JointControl, 'joint_control', self.control_joint_callback)

        # Simulate joint positions
        self.joint_positions = {
            'arm_joint': 0.0,
            'leg_joint': 0.0,
            'head_joint': 0.0
        }

    def control_joint_callback(self, request, response):
        self.get_logger().info(f'Received request to move {request.joint_name} to {request.target_position}')

        # Simulate joint movement
        if request.joint_name in self.joint_positions:
            # Simulate movement time
            time.sleep(0.5)
            self.joint_positions[request.joint_name] = request.target_position

            response.success = True
            response.message = f'Successfully moved {request.joint_name} to {request.target_position}'
            self.get_logger().info(f'Moved {request.joint_name} to {request.target_position}')
        else:
            response.success = False
            response.message = f'Joint {request.joint_name} not found'
            self.get_logger().error(f'Joint {request.joint_name} not found')

        return response

def main(args=None):
    rclpy.init(args=args)
    joint_control_service = JointControlService()

    try:
        rclpy.spin(joint_control_service)
    except KeyboardInterrupt:
        pass
    finally:
        joint_control_service.destroy_node()
        rclpy.shutdown()

if __name__ == '__main__':
    main()
```

#### 3. Create the AI Agent Client

```python
import rclpy
from rclpy.node import Node
from your_package.srv import JointControl  # Assuming you have the service definition

class AIAgentClient(Node):
    def __init__(self):
        super().__init__('ai_agent_client')
        self.cli = self.create_client(JointControl, 'joint_control')
        while not self.cli.wait_for_service(timeout_sec=1.0):
            self.get_logger().info('Service not available, waiting again...')

        # Timer to trigger AI decisions
        self.timer = self.create_timer(2.0, self.ai_decision_callback)

    def ai_decision_callback(self):
        # Simple AI decision making
        request = JointControl.Request()
        request.joint_name = 'arm_joint'
        request.target_position = 1.5  # radians

        self.future = self.cli.call_async(request)
        self.future.add_done_callback(self.service_response_callback)

    def service_response_callback(self, future):
        try:
            response = future.result()
            if response.success:
                self.get_logger().info(f'Success: {response.message}')
            else:
                self.get_logger().error(f'Failed: {response.message}')
        except Exception as e:
            self.get_logger().error(f'Service call failed: {e}')

def main(args=None):
    rclpy.init(args=args)
    ai_agent_client = AIAgentClient()

    try:
        rclpy.spin(ai_agent_client)
    except KeyboardInterrupt:
        pass
    finally:
        ai_agent_client.destroy_node()
        rclpy.shutdown()

if __name__ == '__main__':
    main()
```
