#### 3. Create the Collision Detection Node

```python
import rclpy
from rclpy.node import Node
from sensor_msgs.msg import LaserScan
from geometry_msgs.msg import Twist
from nav_msgs.msg import Odometry
import numpy as np

class CollisionDetector(Node):
    def __init__(self):
        super().__init__('collision_detector')

        # Subscribers
        self.lidar_sub = self.create_subscription(
            LaserScan, '/lidar/scan', self.lidar_callback, 10)
        self.odom_sub = self.create_subscription(
            Odometry, '/odom', self.odom_callback, 10)

        # Publishers
        self.cmd_pub = self.create_publisher(Twist, '/cmd_vel', 10)
        self.collision_pub = self.create_publisher(
            Twist, '/collision_response', 10)  # For collision alerts

        # Robot state
        self.current_velocity = Twist()
        self.collision_threshold = 0.5  # meters
        self.safety_margin = 0.8  # meters

        self.get_logger().info('Collision Detector initialized')

    def lidar_callback(self, msg):
        # Convert to numpy array and filter valid readings
        ranges = np.array(msg.ranges)
        valid_ranges = ranges[np.isfinite(ranges)]

        if len(valid_ranges) == 0:
            return

        # Find minimum distance
        min_distance = np.min(valid_ranges)

        # Check for potential collision
        if min_distance < self.collision_threshold:
            self.get_logger().warn(f'Collision imminent! Distance: {min_distance:.2f}m')
            self.execute_collision_avoidance()
        elif min_distance < self.safety_margin:
            self.get_logger().info(f'Obstacle detected: {min_distance:.2f}m')
            self.slow_down_approach()
        else:
            self.get_logger().info(f'Path clear: {min_distance:.2f}m')

    def odom_callback(self, msg):
        # Store current velocity for collision response
        self.current_velocity = msg.twist.twist

    def execute_collision_avoidance(self):
        """Emergency stop and turn to avoid collision"""
        cmd_msg = Twist()
        cmd_msg.linear.x = 0.0  # Stop linear motion
        cmd_msg.angular.z = 0.5  # Turn right to avoid obstacle

        self.cmd_pub.publish(cmd_msg)
        self.collision_pub.publish(cmd_msg)

    def slow_down_approach(self):
        """Slow down when approaching obstacles"""
        cmd_msg = Twist()
        cmd_msg.linear.x = max(0.1, self.current_velocity.linear.x * 0.5)  # Reduce speed by half
        cmd_msg.angular.z = self.current_velocity.angular.z  # Maintain turning

        self.cmd_pub.publish(cmd_msg)

def main(args=None):
    rclpy.init(args=args)
    collision_detector = CollisionDetector()

    try:
        rclpy.spin(collision_detector)
    except KeyboardInterrupt:
        pass
    finally:
        collision_detector.destroy_node()
        rclpy.shutdown()

if __name__ == '__main__':
    main()
```

```csharp
using UnityEngine;
using System.Collections;

public class UnityInteractionScene : MonoBehaviour
{
    [Header("Robot Configuration")]
    public GameObject robot;
    public float robotSpeed = 2.0f;
    public float robotRotationSpeed = 90.0f; // degrees per second

    [Header("Interaction Objects")]
    public GameObject[] interactionObjects;
    public Material highlightedMaterial;
    public Material defaultMaterial;

    [Header("UI Elements")]
    public UnityEngine.UI.Text interactionText;
    public UnityEngine.UI.Slider distanceSlider;

    private Renderer robotRenderer;
    private bool isInteracting = false;
    private GameObject currentInteractionTarget = null;

    void Start()
    {
        // Initialize robot renderer
        robotRenderer = robot.GetComponent<Renderer>();

        // Set up interaction objects with triggers
        SetupInteractionObjects();
    }

    void Update()
    {
        HandleRobotMovement();
        CheckForInteractions();
    }

    void SetupInteractionObjects()
    {
        foreach (GameObject obj in interactionObjects)
        {
            // Add collider if not present
            if (obj.GetComponent<Collider>() == null)
            {
                obj.AddComponent<BoxCollider>();
            }

            // Make it a trigger
            obj.GetComponent<Collider>().isTrigger = true;

            // Store original material
            Renderer objRenderer = obj.GetComponent<Renderer>();
            if (objRenderer != null)
            {
                defaultMaterial = objRenderer.material;
            }
        }
    }

    void HandleRobotMovement()
    {
        // Simple movement controls
        float horizontal = Input.GetAxis("Horizontal");
        float vertical = Input.GetAxis("Vertical");

        Vector3 movement = new Vector3(horizontal, 0, vertical) * robotSpeed * Time.deltaTime;
        robot.transform.Translate(movement, Space.World);

        // Rotate robot based on movement direction
        if (movement != Vector3.zero)
        {
            Quaternion newRotation = Quaternion.LookRotation(movement);
            robot.transform.rotation = Quaternion.RotateTowards(
                robot.transform.rotation,
                newRotation,
                robotRotationSpeed * Time.deltaTime
            );
        }
    }

    void CheckForInteractions()
    {
        // Check distance to interaction objects
        foreach (GameObject obj in interactionObjects)
        {
            float distance = Vector3.Distance(robot.transform.position, obj.transform.position);

            // Update UI
            if (distanceSlider != null)
            {
                distanceSlider.value = Mathf.Clamp01(distance / 5.0f); // Normalize to 0-5m range
            }

            // Check if close enough for interaction
            if (distance < 2.0f)
            {
                HighlightObject(obj);

                if (interactionText != null)
                {
                    interactionText.text = $"Near {obj.name}\nPress E to interact";
                }

                // Check for interaction key
                if (Input.GetKeyDown(KeyCode.E))
                {
                    InteractWithObject(obj);
                }
            }
            else
            {
                RemoveHighlight(obj);
            }
        }
    }

    void HighlightObject(GameObject obj)
    {
        Renderer objRenderer = obj.GetComponent<Renderer>();
        if (objRenderer != null && highlightedMaterial != null)
        {
            objRenderer.material = highlightedMaterial;
        }
    }

    void RemoveHighlight(GameObject obj)
    {
        Renderer objRenderer = obj.GetComponent<Renderer>();
        if (objRenderer != null && defaultMaterial != null)
        {
            objRenderer.material = defaultMaterial;
        }
    }

    void InteractWithObject(GameObject obj)
    {
        if (isInteracting) return;

        isInteracting = true;
        currentInteractionTarget = obj;

        // Change robot color to indicate interaction
        if (robotRenderer != null)
        {
            robotRenderer.material.color = Color.green;
        }

        // Perform interaction based on object type
        StartCoroutine(PerformInteraction(obj));
    }

    IEnumerator PerformInteraction(GameObject obj)
    {
        // Example interaction: make object pulse
        Renderer objRenderer = obj.GetComponent<Renderer>();
        if (objRenderer != null)
        {
            Color originalColor = objRenderer.material.color;
            float pulseSpeed = 2.0f;

            // Pulse animation
            for (float t = 0; t < 2.0f; t += Time.deltaTime)
            {
                float pulse = Mathf.PingPong(t * pulseSpeed, 1.0f);
                objRenderer.material.color = Color.Lerp(originalColor, Color.yellow, pulse);
                yield return null;
            }

            objRenderer.material.color = originalColor;
        }

        // Reset interaction state
        yield return new WaitForSeconds(0.5f);
        if (robotRenderer != null)
        {
            robotRenderer.material.color = Color.blue; // Back to original color
        }
        isInteracting = false;
        currentInteractionTarget = null;
    }
}
```

#### 2. Physics-Based Trigger System

```csharp
using UnityEngine;

public class PhysicsTrigger : MonoBehaviour
{
    [Header("Trigger Settings")]
    public float detectionRadius = 2.0f;
    public LayerMask detectionLayer = -1; // All layers
    public bool useContinuousDetection = false;

    [Header("Visual Feedback")]
    public GameObject detectionIndicator;
    public Material activeMaterial;
    public Material inactiveMaterial;

    [Header("Events")]
    public UnityEngine.Events.UnityEvent onObjectDetected;
    public UnityEngine.Events.UnityEvent onObjectLost;

    private Collider[] detectedObjects = new Collider[10];
    private int lastDetectionCount = 0;

    void Start()
    {
        if (detectionIndicator != null)
        {
            detectionIndicator.SetActive(false);
        }
    }

    void Update()
    {
        DetectObjects();
    }

    void DetectObjects()
    {
        // Use Physics.OverlapSphere to detect objects in range
        int detectionCount = Physics.OverlapSphereNonAlloc(
            transform.position,
            detectionRadius,
            detectedObjects,
            detectionLayer,
            useContinuousDetection ? QueryTriggerInteraction.UseGlobal : QueryTriggerInteraction.Ignore
        );

        // Check if we detected new objects or lost old ones
        if (detectionCount > 0 && lastDetectionCount == 0)
        {
            // New object detected
            onObjectDetected?.Invoke();
            ShowDetectionIndicator(true);
        }
        else if (detectionCount == 0 && lastDetectionCount > 0)
        {
            // All objects lost
            onObjectLost?.Invoke();
            ShowDetectionIndicator(false);
        }

        lastDetectionCount = detectionCount;
    }

    void ShowDetectionIndicator(bool show)
    {
        if (detectionIndicator != null)
        {
            detectionIndicator.SetActive(show);
        }

        // Change material based on detection state
        Renderer triggerRenderer = GetComponent<Renderer>();
        if (triggerRenderer != null)
        {
            triggerRenderer.material = show ? activeMaterial : inactiveMaterial;
        }
    }

    // For visualization in the editor
    void OnDrawGizmosSelected()
    {
        Gizmos.color = Color.yellow;
        Gizmos.DrawWireSphere(transform.position, detectionRadius);
    }
}
```

```csharp
        sample.robotState.rotation[0] = transform.rotation.x;
        sample.robotState.rotation[1] = transform.rotation.y;
        sample.robotState.rotation[2] = transform.rotation.z;
        sample.robotState.rotation[3] = transform.rotation.w;

        // Collect environment states
        sample.environmentStates = new List<EnvironmentState>();
        GameObject[] environmentObjects = GameObject.FindGameObjectsWithTag("Environment");
        foreach (GameObject obj in environmentObjects)
        {
            EnvironmentState envState = new EnvironmentState();
            envState.name = obj.name;
            envState.position[0] = obj.transform.position.x;
            envState.position[1] = obj.transform.position.y;
            envState.position[2] = obj.transform.position.z;
            envState.objectType = obj.tag;
            sample.environmentStates.Add(envState);
        }

        // Collect sensor readings (simulated)
        sample.sensorReadings = new List<SensorReading>();
        SensorReading lidarReading = new SensorReading();
        lidarReading.sensorType = "lidar";
        lidarReading.data = GenerateSimulatedLidarData();
        lidarReading.timestamp = Time.time;
        sample.sensorReadings.Add(lidarReading);

        // Save sample metadata
        string json = JsonUtility.ToJson(sample, true);
        string filename = Path.Combine(sessionPath, "metadata", $"sample_{sampleCount:D6}.json");
        File.WriteAllText(filename, json);

        // Capture and save image
        string imageFilename = Path.Combine(sessionPath, "images", $"image_{sampleCount:D6}.png");
        CaptureScreenImage(imageFilename);

        sampleCount++;
        Debug.Log($"Collected training sample {sampleCount}");
    }

    float[] GenerateSimulatedLidarData()
    {
        // Generate simulated 360-degree LiDAR data
        float[] lidarData = new float[360];
        for (int i = 0; i < 360; i++)
        {
            // Simulate distances based on environment
            float angle = i * Mathf.Deg2Rad;
            Vector3 direction = new Vector3(Mathf.Cos(angle), 0, Mathf.Sin(angle));

            // Raycast to find distance to nearest object
            RaycastHit hit;
            if (Physics.Raycast(transform.position, direction, out hit, 10.0f))
            {
                lidarData[i] = hit.distance;
            }
            else
            {
                lidarData[i] = 10.0f; // Max range
            }
        }
        return lidarData;
    }

    void CaptureScreenImage(string filename)
    {
        // Create a texture to read pixels from
        Texture2D texture = new Texture2D(Screen.width, Screen.height, TextureFormat.RGB24, false);

        // Read screen contents into the texture
        texture.ReadPixels(new Rect(0, 0, Screen.width, Screen.height), 0, 0);
        texture.Apply();

        // Encode texture into PNG
        byte[] bytes = texture.EncodeToPNG();
        File.WriteAllBytes(filename, bytes);

        // Clean up
        Destroy(texture);
    }
}
```
