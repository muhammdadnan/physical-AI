# 3D Perception Pipelines

## Introduction to 3D Perception

3D perception is fundamental to humanoid robotics, enabling robots to understand and navigate their environment in three dimensions. Isaac ROS provides powerful GPU-accelerated tools for 3D perception, including stereo vision, depth estimation, point cloud processing, and spatial reasoning capabilities.

## Stereo Vision and Depth Estimation

### Stereo Vision Fundamentals

Stereo vision uses two cameras to estimate depth by finding corresponding points in the left and right images. The disparity between corresponding points is inversely proportional to the depth.

class StereoDepthNode(Node):

        self.right_sub = message_filters.Subscriber(self, Image, '/stereo/right/image_rect')
        self.left_info_sub = message_filters.Subscriber(self, CameraInfo, '/stereo/left/camera_info')
        self.right_info_sub = message_filters.Subscriber(self, CameraInfo, '/stereo/right/camera_info')

        # Synchronize stereo images
        self.stereo_sync = message_filters.ApproximateTimeSynchronizer(
            [self.left_sub, self.right_sub, self.left_info_sub, self.right_info_sub],

        self.pc_pub = self.create_publisher(PointCloud2, '/stereo/pointcloud', 10)

        # Stereo parameters
        self.stereo_matcher = cv2.StereoSGBM_create(
            minDisparity=0,
            numDisparities=128,
            blockSize=5,
            P1=8 * 3 * 5**2,
            P2=32 * 3 * 5**2,
            disp12MaxDiff=1,
            uniquenessRatio=15,
            speckleWindowSize=0,
            speckleRange=2,
            preFilterCap=63,
            mode=cv2.STEREO_SGBM_MODE_SGBM_3WAY
        )

        self.q_matrix = None  # Will be computed from camera calibration

    def stereo_callback(self, left_msg, right_msg, left_info, right_info):
        """

        """

            # Publish depth image
            depth_msg = self.bridge.cv2_to_imgmsg(depth_image, "32FC1")
            depth_msg.header = left_msg.header
            self.depth_pub.publish(depth_msg)

            # Generate point cloud
            pointcloud = self.generate_pointcloud(left_cv, depth_image, left_info)

        except Exception as e:
            self.get_logger().error(f'Stereo processing error: {e}')

    def disparity_to_depth(self, disparity):
        """
        Convert disparity image to depth image
        """
        # Need to compute Q matrix from stereo calibration
        # Q matrix transforms disparity to 3D coordinates
        if self.q_matrix is None:
            self.compute_q_matrix(left_info, right_info)

        # For now, use a simplified approach
        # Depth = focal_length * baseline / disparity
        focal_length = 525.0  # Example focal length in pixels
        baseline = 0.1  # Example baseline in meters (10cm)

        # Avoid division by zero
        disparity[disparity == 0] = 0.01
        depth = (focal_length * baseline) / disparity
        depth[depth > 10.0] = 0  # Limit depth range

        return depth

    def compute_q_matrix(self, left_info, right_info):
        """
        Compute Q matrix for disparity to depth conversion
        """
        # Extract stereo parameters from camera calibration
        # This is a simplified version - in practice, you'd use the full calibration
        fx = left_info.P[0]  # Focal length x
        fy = left_info.P[5]  # Focal length y
        cx = left_info.P[2]  # Principal point x
        cy = left_info.P[6]  # Principal point y
        tx = left_info.P[3]  # Baseline * fx

        # Q matrix for cv2.reprojectImageTo3D
        self.q_matrix = np.array([
            [1, 0, 0, -cx],
            [0, 1, 0, -cy],
            [0, 0, 0, fx],
            [0, 0, -1/tx, 0]
        ])

        """
        Process incoming point cloud from Isaac ROS stereo reconstruction
        """
        try:
            # Convert PointCloud2 to numpy array for processing

            # Apply GPU-accelerated processing

            # Segment objects in point cloud

            # Publish results
            self.publish_segmented_objects(segmented_objects)

            self.get_logger().error(f'Point cloud processing error: {e}')

    def pointcloud2_to_array(self, cloud):
        """
        Convert PointCloud2 message to numpy array
        """
        import sensor_msgs.point_cloud2 as pc2
        points_list = []

        for point in pc2.read_points(cloud, field_names=("x", "y", "z"), skip_nans=True):
            points_list.append([point[0], point[1], point[2]])

        return np.array(points_list)

    def process_pointcloud_gpu(self, points):
        """
        Process point cloud using GPU acceleration
        """
        import cupy as cp

        # Transfer to GPU
        gpu_points = cp.asarray(points)

        # Apply transformations or filtering on GPU
        # Example: remove points too far away
        valid_mask = cp.sqrt(cp.sum(gpu_points**2, axis=1)) < 5.0  # 5m range
        filtered_points = gpu_points[valid_mask]

        # Transfer back to CPU
        return cp.asnumpy(filtered_points)

    def segment_objects(self, points):
        """
        Segment objects in the point cloud
        """
        from sklearn.cluster import DBSCAN

        # Use DBSCAN clustering to segment objects
        clustering = DBSCAN(eps=0.2, min_samples=10)  # 20cm radius, 10 points minimum
        labels = clustering.fit_predict(points)

        # Group points by cluster
        objects = {}
        for label in np.unique(labels):
            if label == -1:  # Noise points
                continue
            object_points = points[labels == label]
            objects[label] = {
                'points': object_points,
                'centroid': np.mean(object_points, axis=0),
                'bbox': self.compute_bounding_box(object_points)
            }

        return objects

    def compute_bounding_box(self, points):
        """
        Compute axis-aligned bounding box for a point cluster
        """
        min_coords = np.min(points, axis=0)
        max_coords = np.max(points, axis=0)
        center = (min_coords + max_coords) / 2
        size = max_coords - min_coords

        return {
            'center': center,
            'size': size,
            'min': min_coords,
            'max': max_coords
        }
```

class PointCloudProcessor(Node):
    def __init__(self):
        super().__init__('pointcloud_processor')

        self.pc_sub = self.create_subscription(
            PointCloud2,
            '/input/pointcloud',
            self.pc_callback,
            10
        )

        self.filtered_pub = self.create_publisher(
            PointCloud2,
            '/filtered/pointcloud',
            10
        )

        self.ground_pub = self.create_publisher(
            PointCloud2,
            '/ground/pointcloud',
            10
        )

        self.obstacles_pub = self.create_publisher(
            PointCloud2,
            '/obstacles/pointcloud',
            10
        )

    def pc_callback(self, msg):
        """
        Process incoming point cloud with filtering and segmentation
        """
        try:
            points = self.pointcloud2_to_array(msg)

            # Remove NaN values
            valid_mask = np.isfinite(points).all(axis=1)
            points = points[valid_mask]

            # Apply voxel grid filtering for downsampling
            filtered_points = self.voxel_grid_filter(points, leaf_size=0.05)  # 5cm voxels

            # Separate ground plane from obstacles
            ground_points, obstacle_points = self.separate_ground_obstacles(filtered_points)

            # Publish results
            self.publish_pointcloud(self.filtered_pub, filtered_points, msg.header)
            self.publish_pointcloud(self.ground_pub, ground_points, msg.header)
            self.publish_pointcloud(self.obstacles_pub, obstacle_points, msg.header)

        except Exception as e:
            self.get_logger().error(f'Point cloud processing error: {e}')

    def voxel_grid_filter(self, points, leaf_size=0.05):
        """
        Apply voxel grid filtering to downsample point cloud
        """
        # Compute voxel indices
        min_coords = np.min(points, axis=0)
        voxel_indices = ((points - min_coords) / leaf_size).astype(int)

        # Group points by voxel
        voxel_dict = {}
        for i, voxel_idx in enumerate(voxel_indices):
            voxel_key = tuple(voxel_idx)
            if voxel_key not in voxel_dict:
                voxel_dict[voxel_key] = []
            voxel_dict[voxel_key].append(points[i])

        # Take centroid of each voxel
        filtered_points = []
        for voxel_points in voxel_dict.values():
            centroid = np.mean(voxel_points, axis=0)
            filtered_points.append(centroid)

        return np.array(filtered_points)

    def separate_ground_obstacles(self, points):
        """
        Separate ground plane from obstacles using RANSAC
        """
        from sklearn.linear_model import RANSACRegressor

        # Prepare data for RANSAC (only x, z for 2D plane fitting)
        X = points[:, [0, 2]]  # x, z coordinates
        y = points[:, 1]       # y coordinate (height)

        # Fit ground plane

        ransac.fit(X, y)

        # Classify points

        inlier_mask = np.abs(y - y_pred) < 0.1  # 10cm tolerance

class SpatialReasoningNode(Node):
    def __init__(self):
        super().__init__('spatial_reasoning_node')

        self.pointcloud_sub = self.create_subscription(
            PointCloud2,
            '/processed/pointcloud',
            self.pointcloud_callback,
            10
        )

        self.occupancy_grid_pub = self.create_publisher(
            'nav_msgs/msg/OccupancyGrid',
            '/spatial_map',
            10
        )

        self.spatial_query_service = self.create_service(
            'isaac_ros_interfaces/srv/SpatialQuery',
            'query_spatial',
            self.spatial_query_callback
        )

        # Spatial map parameters
        self.grid_resolution = 0.1  # 10cm resolution
        self.grid_width = 200       # 20m x 20m grid
        self.grid_height = 200
        self.grid_origin = [-10.0, -10.0]  # Centered at robot

        # Initialize occupancy grid
        self.occupancy_grid = np.zeros((self.grid_height, self.grid_width), dtype=np.int8)

    def pointcloud_callback(self, msg):
        """
        Update occupancy grid from point cloud data
        """
        points = self.pointcloud2_to_array(msg)

        # Convert world coordinates to grid coordinates
        grid_points = self.world_to_grid(points)

        # Update occupancy grid
        for point in grid_points:
            x, y = int(point[0]), int(point[1])
            if 0 <= x < self.grid_width and 0 <= y < self.grid_height:
                # Mark as occupied
                self.occupancy_grid[y, x] = 100  # 100 = definitely occupied

        # Publish updated grid
        self.publish_occupancy_grid()

    def world_to_grid(self, world_points):
        """
        Convert world coordinates to grid coordinates
        """
        grid_points = np.zeros_like(world_points)
        grid_points[:, 0] = (world_points[:, 0] - self.grid_origin[0]) / self.grid_resolution
        grid_points[:, 1] = (world_points[:, 1] - self.grid_origin[1]) / self.grid_resolution
        return grid_points

    def spatial_query_callback(self, request, response):
        """
        Handle spatial reasoning queries
        """
        x, y = request.x, request.y

        # Convert to grid coordinates
        grid_x = int((x - self.grid_origin[0]) / self.grid_resolution)
        grid_y = int((y - self.grid_origin[1]) / self.grid_resolution)

        # Check if coordinates are valid
        if 0 <= grid_x < self.grid_width and 0 <= grid_y < self.grid_height:
            occupancy = self.occupancy_grid[grid_y, grid_x]
            response.occupancy = occupancy
            response.is_free = occupancy < 50  # Below 50% is considered free
        else:
            response.occupancy = -1  # Unknown
            response.is_free = False

        return response

    def publish_occupancy_grid(self):
        """
        Publish occupancy grid as ROS message
        """
        from nav_msgs.msg import OccupancyGrid
        from geometry_msgs.msg import Pose

        grid_msg = OccupancyGrid()
        grid_msg.header.stamp = self.get_clock().now().to_msg()
        grid_msg.header.frame_id = "map"

        grid_msg.info.resolution = self.grid_resolution
        grid_msg.info.width = self.grid_width
        grid_msg.info.height = self.grid_height

        # Set origin (position of grid lower-left corner in map frame)
        origin = Pose()
        origin.position.x = self.grid_origin[0]
        origin.position.y = self.grid_origin[1]
        grid_msg.info.origin = origin

        # Flatten grid for message
        grid_msg.data = self.occupancy_grid.flatten().tolist()

        self.occupancy_grid_pub.publish(grid_msg)
```

class OptimizedPerceptionNode(Node):
    def __init__(self):
        super().__init__('optimized_perception_node')

        # GPU memory management
        import cupy as cp
        self.gpu_memory_pool = cp.cuda.MemoryPool()
        cp.cuda.set_allocator(self.gpu_memory_pool.malloc)

        # Set memory limit
        device = cp.cuda.Device()
        memory_limit = int(device.mem_info[0] * 0.8)  # Use 80% of available memory
        self.gpu_memory_pool.set_limit(size=memory_limit)

        # Initialize GPU-based processing modules
        self.initialize_gpu_modules()

    def initialize_gpu_modules(self):
        """
        Initialize GPU-based processing modules
        """
        import cupy as cp

        # Pre-allocate GPU arrays for common operations
        self.gpu_image_buffer = cp.empty((480, 640, 3), dtype=cp.uint8)
        self.gpu_depth_buffer = cp.empty((480, 640), dtype=cp.float32)
        self.gpu_point_buffer = cp.empty((100000, 3), dtype=cp.float32)

    def process_frame_optimized(self, image_msg, pointcloud_msg):
        """
        Optimized frame processing with GPU memory reuse
        """
        import cupy as cp

        try:
            # Convert image to GPU memory
            cpu_image = self.bridge.imgmsg_to_cv2(image_msg, "bgr8")
            self.gpu_image_buffer.set(cpu_image)

            # Process on GPU
            processed_gpu = self.gpu_process(self.gpu_image_buffer)

            # Transfer result back to CPU
            result_cpu = cp.asnumpy(processed_gpu)

            # Publish result
            result_msg = self.bridge.cv2_to_imgmsg(result_cpu, "bgr8")
            result_msg.header = image_msg.header
            self.result_pub.publish(result_msg)

        except cp.cuda.memory.OutOfMemoryError:

## Best Practices for 3D Perception

1. **Multi-sensor Fusion**: Combine data from multiple sensors (cameras, LiDAR, depth sensors)
2. **GPU Acceleration**: Leverage GPU for computationally intensive operations
3. **Real-time Processing**: Optimize for real-time performance requirements
4. **Robust Filtering**: Implement robust filtering to handle sensor noise
5. **Coordinate Systems**: Maintain consistent coordinate system transformations
6. **Memory Management**: Efficiently manage GPU and CPU memory
7. **Validation**: Continuously validate perception results against ground truth
