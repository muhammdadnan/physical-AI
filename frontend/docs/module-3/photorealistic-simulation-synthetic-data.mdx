# Photorealistic Simulation & Synthetic Data

## Introduction to Photorealistic Simulation

Photorealistic simulation in Isaac Sim leverages NVIDIA's advanced rendering capabilities to create highly realistic environments for robotics training. This technology enables the generation of synthetic data that closely mimics real-world conditions, making it invaluable for training AI models without requiring physical data collection.

## Rendering Modes in Isaac Sim

### USD Render Modes

Isaac Sim supports multiple rendering modes for different use cases:

#### 1. Kit Renderer (Default)
- Fast, real-time rendering
- Good for interactive simulation
- Suitable for most robotics applications

#### 2. Path Tracing
- Physically accurate lighting simulation
- High-quality, photorealistic output
- Slower but more realistic than Kit renderer

#### 3. RTX Renderer
- Hardware-accelerated ray tracing
- Balance of quality and performance
- Requires RTX-capable GPU

## Creating Synthetic Datasets

### RGB Image Generation

```python
import omni
from omni.isaac.core.utils.viewports import set_camera_view
from omni.isaac.sensor import Camera
import numpy as np
import cv2

def setup_synthetic_camera(world, position, target):
    """
    Set up a camera for synthetic RGB data generation
    """
    # Create camera prim
    camera_prim_path = "/World/SyntheticCamera"

    # Add camera to the scene
    camera = Camera(
        prim_path=camera_prim_path,
        frequency=30,  # 30 Hz
        resolution=(640, 480)
    )

    # Position the camera
    camera.set_world_pose(position, target)

    return camera

def capture_synthetic_images(camera, output_dir, num_images=100):
    """
    Capture synthetic RGB images for dataset generation
    """
    import os

    # Create output directory
    os.makedirs(output_dir, exist_ok=True)

    for i in range(num_images):
        # Step the world to update camera data
        world.step(render=True)

        # Get RGB data
        rgb_data = camera.get_rgb()

        if rgb_data is not None:
            # Convert to OpenCV format
            rgb_image = cv2.cvtColor(rgb_data, cv2.COLOR_RGBA2BGR)

            # Save image
            image_path = f"{output_dir}/rgb_{i:06d}.png"
            cv2.imwrite(image_path, rgb_image)

            print(f"Saved RGB image: {image_path}")
```

## Advanced Synthetic Data Techniques

### Domain Randomization

Domain randomization is a technique to increase the robustness of models trained on synthetic data:

```python
import random
from pxr import UsdShade, Gf

def apply_domain_randomization(world, objects):
    """
    Apply randomization to object properties for domain randomization
    """
    for obj in objects:
        # Randomize material properties
        randomize_material(obj)

        # Randomize lighting
        randomize_lighting()

        # Randomize camera parameters
        randomize_camera_params()

def randomize_material(obj):
    """
    Randomize material properties of an object
    """
    # Randomize color
    base_color = Gf.Vec3f(
        random.uniform(0.1, 1.0),
        random.uniform(0.1, 1.0),
        random.uniform(0.1, 1.0)
    )

    # Randomize roughness
    roughness = random.uniform(0.1, 0.9)

    # Randomize metallic
    metallic = random.uniform(0.0, 0.5)

    # Apply changes to material
    material_path = obj.GetPath().AppendPath("Material")
    material = UsdShade.Material.Define(world.stage, material_path)

    # Set material properties
    pbr_shader = UsdShade.Shader.Define(world.stage, material_path.AppendPath("PBRShader"))
    pbr_shader.CreateIdAttr("OmniPBR")

    # Set properties
    pbr_shader.CreateInput("diffuse_color", Sdf.ValueTypeNames.Color3f).Set(base_color)
    pbr_shader.CreateInput("roughness", Sdf.ValueTypeNames.Float).Set(roughness)
    pbr_shader.CreateInput("metallic", Sdf.ValueTypeNames.Float).Set(metallic)

def randomize_lighting():
    """
    Randomize lighting conditions
    """
    # Get all lights in the scene
    lights = world.scene.get_objects_by_type("DistantLight")

    for light in lights:
        # Randomize intensity
        intensity = random.uniform(500, 1500)
        light.intensity = intensity

        # Randomize color temperature
        color_temp = random.uniform(4000, 7000)  # Kelvin
        # Convert to RGB approximation
        rgb_color = kelvin_to_rgb(color_temp)
        light.color = rgb_color

def kelvin_to_rgb(kelvin):
    """
    Convert Kelvin temperature to RGB color
    """
    temp = kelvin / 100
    red, green, blue = 0, 0, 0

    # Red calculation
    if temp <= 66:
        red = 255
    else:
        red = temp - 60
        red = 329.698727446 * (red ** -0.1332047592)
        red = max(0, min(255, red))

    # Green calculation
    if temp <= 66:
        green = temp
        green = 99.4708025861 * math.log(green) - 161.1195681661
    else:
        green = temp - 60
        green = 288.1221695283 * (green ** -0.0755148492)
    green = max(0, min(255, green))

    # Blue calculation
    if temp >= 66:
        blue = 255
    elif temp <= 19:
        blue = 0
    else:
        blue = temp - 10
        blue = 138.5177312231 * math.log(blue) - 305.0447927307
        blue = max(0, min(255, blue))

    return (red/255.0, green/255.0, blue/255.0)
```

```python
def create_training_split(data_dir, train_ratio=0.7, val_ratio=0.2, test_ratio=0.1):
    """
    Create train/validation/test splits for the dataset
    """
    import os
    import random
    from shutil import copy2

    # Get all data files
    rgb_files = [f for f in os.listdir(f"{data_dir}/rgb") if f.endswith('.png')]

    # Shuffle files
    random.shuffle(rgb_files)

    # Calculate split sizes
    total_files = len(rgb_files)
    train_end = int(total_files * train_ratio)
    val_end = train_end + int(total_files * val_ratio)

    # Create splits
    splits = {
        'train': rgb_files[:train_end],
        'val': rgb_files[train_end:val_end],
        'test': rgb_files[val_end:]
    }

    # Create split directories
    for split_name in splits:
        os.makedirs(f"{data_dir}/{split_name}/rgb", exist_ok=True)
        os.makedirs(f"{data_dir}/{split_name}/depth", exist_ok=True)
        os.makedirs(f"{data_dir}/{split_name}/segmentation", exist_ok=True)

    # Copy files to appropriate splits
    for split_name, files in splits.items():
        for file in files:
            # Copy RGB
            src_rgb = f"{data_dir}/rgb/{file}"
            dst_rgb = f"{data_dir}/{split_name}/rgb/{file}"
            copy2(src_rgb, dst_rgb)

            # Copy corresponding depth and segmentation
            base_name = file.replace('.png', '')
            for data_type in ['depth', 'segmentation']:
                src_data = f"{data_dir}/{data_type}/{base_name}.png"
                dst_data = f"{data_dir}/{split_name}/{data_type}/{base_name}.png"
                if os.path.exists(src_data):
                    copy2(src_data, dst_data)

    print(f"Dataset split: {len(splits['train'])} train, {len(splits['val'])} val, {len(splits['test'])} test")
```

## Quality Assurance for Synthetic Data

### Data Validation

```python
def validate_synthetic_data(data_dir):
    """
    Validate the quality and consistency of synthetic data
    """
    import os
    import cv2
    import numpy as np

    validation_results = {
        'total_images': 0,
        'valid_images': 0,
        'average_brightness': 0,
        'brightness_range_ok': True,
        'resolution_consistent': True,
        'no_corrupted_files': True
    }

    rgb_dir = f"{data_dir}/rgb"
    rgb_files = [f for f in os.listdir(rgb_dir) if f.endswith('.png')]

    total_brightness = 0

    for file in rgb_files:
        validation_results['total_images'] += 1

        # Load image
        img_path = os.path.join(rgb_dir, file)
        try:
            img = cv2.imread(img_path)
            if img is None:
                print(f"Corrupted image: {file}")
                validation_results['no_corrupted_files'] = False
                continue

            # Check resolution consistency
            if img.shape != (480, 640, 3):  # Assuming 640x480 RGB
                validation_results['resolution_consistent'] = False

            # Calculate brightness
            gray = cv2.cvtColor(img, cv2.COLOR_BGR2GRAY)
            brightness = np.mean(gray)
            total_brightness += brightness

            # Check if brightness is in reasonable range
            if brightness < 10 or brightness > 245:  # Too dark or too bright
                print(f"Image brightness out of range: {file}, brightness={brightness}")

            validation_results['valid_images'] += 1

        except Exception as e:
            print(f"Error processing {file}: {e}")
            validation_results['no_corrupted_files'] = False

    if validation_results['valid_images'] > 0:
        validation_results['average_brightness'] = total_brightness / validation_results['valid_images']

    return validation_results
```

## Best Practices for Synthetic Data Generation

1. **Diversity**: Ensure scenes include various lighting conditions, object arrangements, and environments
2. **Realism**: Use physically-based materials and lighting for realistic appearance
3. **Annotation Quality**: Maintain consistent and accurate annotations across all data
4. **Domain Randomization**: Apply randomization to improve model generalization
5. **Validation**: Regularly validate data quality and consistency
6. **Documentation**: Keep detailed records of generation parameters for reproducibility
