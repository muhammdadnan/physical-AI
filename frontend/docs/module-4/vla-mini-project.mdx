# Mini-Project: Voice-Controlled Task Runner

## Project Overview

In this mini-project, we'll implement a complete Vision-Language-Action (VLA) pipeline that allows a humanoid robot to receive voice commands, process them through an LLM cognitive planner, and execute the resulting actions. This project integrates all the components we've learned: voice processing, LLM planning, vision perception, and action execution.

## Project Architecture

### System Components

The voice-controlled task runner consists of several interconnected components:

1. **Voice Input System**: Captures and processes voice commands
2. **Natural Language Interface**: Converts speech to text and understands intent
3. **Cognitive Planning Module**: Translates commands into action sequences
4. **Perception System**: Provides environmental context for planning
5. **Action Execution Pipeline**: Executes the planned actions
6. **Feedback System**: Provides status updates and error handling

import speech_recognition as sr
import threading
import time

class AdvancedVoiceCommandHandler:
    def __init__(self, node):
        self.node = node
        self.recognizer = sr.Recognizer()
        self.microphone = sr.Microphone()

        # Adjust for ambient noise
        with self.microphone as source:
            self.recognizer.adjust_for_ambient_noise(source)

        # Command keywords and patterns
        self.command_patterns = {
            "navigation": ["go to", "navigate to", "move to", "walk to", "go", "move"],
            "manipulation": ["pick up", "get", "take", "grab", "lift", "hold", "place", "put", "set"],
            "fetch": ["bring", "fetch", "get me", "bring me", "go get"],
            "location": ["kitchen", "living room", "bedroom", "office", "dining room", "bathroom"],
            "objects": ["cup", "bottle", "book", "laptop", "phone", "remote", "glass", "plate", "fork", "spoon"]
        }

        # Start listening thread
        self.listening = True
        self.listening_thread = threading.Thread(target=self.listen_continuously)
        self.listening_thread.daemon = True
        self.listening_thread.start()

    def listen_continuously(self):
        """
        Continuously listen for voice commands
        """
        while self.listening:
            try:
                with self.microphone as source:
                    # Listen for audio with timeout
                    audio = self.recognizer.listen(source, timeout=1.0, phrase_time_limit=5.0)

                # Recognize speech
                command = self.recognizer.recognize_google(audio)
                self.node.get_logger().info(f'Recognized: {command}')

                # Publish command
                cmd_msg = String()
                cmd_msg.data = command
                self.node.voice_cmd_pub.publish(cmd_msg)

            except sr.WaitTimeoutError:
                # No speech detected, continue listening
                continue
            except sr.UnknownValueError:
                # Speech not understood
                self.node.get_logger().info('Could not understand audio')
                continue
            except sr.RequestError as e:
                # API error
                self.node.get_logger().error(f'Speech recognition error: {e}')
                time.sleep(1)  # Wait before retrying
            except Exception as e:
                self.node.get_logger().error(f'Listening error: {e}')
                time.sleep(0.1)

    def stop_listening(self):
        """
        Stop the listening process
        """
        self.listening = False

    def validate_command(self, command):
        """
        Validate if command is a valid robot command
        """
        command_lower = command.lower()

        # Check if command contains known action patterns
        has_action = any(pattern in command_lower for pattern in
                        self.command_patterns["navigation"] +
                        self.command_patterns["manipulation"] +
                        self.command_patterns["fetch"])

        # Check if command contains known objects or locations
        has_target = any(obj in command_lower for obj in self.command_patterns["objects"]) or \
                    any(loc in command_lower for loc in self.command_patterns["location"])

        return has_action or has_target

        """
        Estimate world coordinates from image coordinates
        """
        # This would use camera calibration and depth information
        # For now, return a simple relative location
        center_x = (bbox[0] + bbox[2]) / 2
        center_y = (bbox[1] + bbox[3]) / 2

        # Convert to relative position in image
        h_pos = "left" if center_x < 213 else "center" if center_x < 426 else "right"
        v_pos = "top" if center_y < 160 else "center" if center_y < 320 else "bottom"

        return f"{v_pos} {h_pos}"

    def analyze_spatial_context(self, detections):
        """
        Analyze spatial relationships between detected objects
        """
        relationships = []

        for i, obj1 in enumerate(detections):
            for j, obj2 in enumerate(detections[i+1:], i+1):
                relationship = self.calculate_spatial_relationship(obj1, obj2)
                if relationship:
                    relationships.append(relationship)

        return relationships

    def calculate_spatial_relationship(self, obj1, obj2):
        """
        Calculate spatial relationship between two objects
        """
        bbox1 = obj1["bbox"]
        bbox2 = obj2["bbox"]

        center1 = ((bbox1[0] + bbox1[2]) / 2, (bbox1[1] + bbox1[3]) / 2)
        center2 = ((bbox2[0] + bbox2[2]) / 2, (bbox2[1] + bbox2[3]) / 2)

        dx = center2[0] - center1[0]
        dy = center2[1] - center1[1]

        # Determine relative position
        if abs(dx) > abs(dy):
            direction = "left" if dx < 0 else "right"
        else:
            direction = "above" if dy < 0 else "below"

        distance = ((dx**2) + (dy**2))**0.5

        return {
            "objects": [obj1["class"], obj2["class"]],
            "relationship": direction,
            "distance": distance
        }

    def find_object_for_action(self, target_class, context):
        """
        Find the most suitable object for a given action
        """
        if "objects" not in context:
            return None

        # Filter objects by class and suitability
        suitable_objects = [
            obj for obj in context["objects"]
            if obj["class"] == target_class and obj.get("manipulable", False)
        ]

        if not suitable_objects:
            # Check object memory for previously seen objects
            if target_class in self.object_memory:
                # Return the most recently seen object
                recent_objects = sorted(
                    self.object_memory[target_class],
                    key=lambda x: x["timestamp"],
                    reverse=True
                )
                if recent_objects:
                    return recent_objects[0]

            return None

        # Return the object with highest confidence
        return max(suitable_objects, key=lambda x: x["confidence"])
