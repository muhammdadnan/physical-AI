async def run_room_service_workflow():

        user_preferences={
            "service_type": "room_service",
            "priority": "normal",
            "safety_thresholds": {"distance": 0.5, "force": 10.0}
        }
    )

    print(f"Started room service session: {session.id}")

    # Define a complex room service task

    # Execute the task with full VLA integration

async def run_multi_user_interaction_workflow():
    """
    Complete workflow: Humanoid robot interacts with multiple users simultaneously
    """
    capstone_service = CapstoneService()

    # Create multi-user interaction project
    project = await capstone_service.create_capstone_project(
        name="Multi-User Interaction",
        description="Humanoid robot that manages tasks for multiple users",
        configuration={
            "robot_type": "social_humanoid",
            "capabilities": ["navigation", "manipulation", "perception", "voice_control", "user_tracking"],
            "environment": "shared_office",
            "user_capacity": 4
        }
    )

    session = await capstone_service.start_capstone_session(
        project_id=project.id,
        user_preferences={
            "user_priorities": {"user_1": "high", "user_2": "medium", "user_3": "low"},
            "interaction_mode": "courteous",  # "efficient", "courteous", "focused"
            "privacy_settings": {"audio_recording": False, "video_recording": False}
        }
    )

    # Simulated multiple user commands arriving over time
    user_commands = [
        {"user_id": "user_1", "command": "Please bring me my laptop from my desk", "timestamp": time.time()},
        {"user_id": "user_2", "command": "Can you go to the printer and bring my documents?", "timestamp": time.time() + 10},
        {"user_id": "user_3", "command": "Navigate to the conference room and wait", "timestamp": time.time() + 25},
        {"user_id": "user_1", "command": "Actually, bring me a glass of water instead", "timestamp": time.time() + 45}
    ]

    task_results = []
    for cmd_data in user_commands:
        print(f"Processing command from {cmd_data['user_id']}: {cmd_data['command']}")

        task_result = await capstone_service.execute_capstone_task(
            session_id=session.id,
            command=cmd_data['command'],
            task_context={
                "requesting_user": cmd_data['user_id'],
                "request_time": cmd_data['timestamp'],
                "user_priority": get_user_priority(cmd_data['user_id']),  # Simulated function
                "current_user_queue": get_current_user_queue(),  # Simulated function
                "conflicting_requests": get_conflicting_requests(cmd_data['command'])  # Simulated function
            }
        )

        task_results.append({
            "user_id": cmd_data['user_id'],
            "command": cmd_data['command'],
            "result": task_result,
            "processing_time": time.time() - cmd_data['timestamp']
        })

    # Generate multi-user interaction report
    interaction_report = await generate_interaction_report(task_results)

    await capstone_service.end_session(session.id)

    return {
        "project_id": project.id,
        "session_id": session.id,
        "task_results": task_results,
        "interaction_report": interaction_report
    }

def get_user_priority(user_id):
    """Simulated function to get user priority"""
    priorities = {"user_1": "high", "user_2": "medium", "user_3": "low", "user_4": "medium"}
    return priorities.get(user_id, "medium")

def get_current_user_queue():
    """Simulated function to get current user request queue"""
    return ["user_1", "user_2", "user_3"]

def get_conflicting_requests(command):
    """Simulated function to detect conflicting requests"""
    # Simple conflict detection
    if "bring me" in command.lower() and "laptop" in command.lower():
        return ["user_2_print_request"]  # Conflicts with printing task
    return []

async def generate_interaction_report(task_results):
    """Generate a report on multi-user interactions"""
    total_tasks = len(task_results)
    successful_tasks = sum(1 for tr in task_results if tr["result"].success)
    user_satisfaction = calculate_user_satisfaction(task_results)

    return {
        "total_interactions": total_tasks,
        "successful_interactions": successful_tasks,
        "success_rate": successful_tasks / total_tasks if total_tasks > 0 else 0,
        "user_satisfaction_score": user_satisfaction,
        "average_response_time": sum(tr["processing_time"] for tr in task_results) / total_tasks if total_tasks > 0 else 0,
        "user_distribution": count_user_interactions(task_results)
    }

def calculate_user_satisfaction(task_results):
    """Calculate user satisfaction based on task outcomes"""
    # Simplified satisfaction calculation
    successful_per_user = {}
    total_per_user = {}

    for result in task_results:
        user_id = result["user_id"]
        total_per_user[user_id] = total_per_user.get(user_id, 0) + 1
        if result["result"].success:
            successful_per_user[user_id] = successful_per_user.get(user_id, 0) + 1

    satisfaction_scores = {}
    for user_id in total_per_user:
        success_rate = successful_per_user.get(user_id, 0) / total_per_user[user_id]
        satisfaction_scores[user_id] = success_rate

    return satisfaction_scores

def count_user_interactions(task_results):
    """Count interactions per user"""
    user_counts = {}
    for result in task_results:
        user_id = result["user_id"]
        user_counts[user_id] = user_counts.get(user_id, 0) + 1
    return user_counts

async def update_learning_context(context, command, task_result, environment_state):
    """Update learning context based on task outcome"""
    # Add task to history
    context["task_history"].append({
        "command": command,
        "result": task_result.success,
        "execution_time": task_result.execution_time,
        "environment": environment_state,
        "timestamp": time.time()
    })

    # Update success/failure patterns
    if task_result.success:
        context["success_patterns"].append({
            "command_pattern": extract_command_pattern(command),
            "environment_conditions": environment_state,
            "execution_approach": task_result.plan  # Simplified
        })
    else:
        context["failure_patterns"].append({
            "command_pattern": extract_command_pattern(command),
            "environment_conditions": environment_state,
            "failure_reason": task_result.error_message
        })

    return context

    """Extract pattern from command for learning"""

    if "go to" in command.lower():

    elif "bring" in command.lower() or "get" in command.lower():

    elif "find" in command.lower():
        return "search_task"
    else:
        return "general_task"

            }

    await capstone_service.end_session(session.id)

async def analyze_recovery_actions(execution_log):
    """Analyze recovery actions taken during task execution"""
    recovery_actions = []

    if execution_log:
        for log_entry in execution_log:
            if isinstance(log_entry, dict):
                # Look for recovery-related keywords in the log
                entry_str = str(log_entry).lower()
                if any(keyword in entry_str for keyword in ["recovery", "fallback", "alternative", "retry", "error"]):
                    recovery_actions.append(log_entry)

    return recovery_actions
