# LLM Cognitive Planning

## Introduction to LLM-Based Cognitive Planning

Large Language Models (LLMs) have emerged as powerful tools for cognitive planning in robotics. They can translate high-level natural language commands into executable action sequences, enabling robots to understand complex tasks and break them down into manageable steps. This cognitive planning capability is essential for Vision-Language-Action (VLA) systems that bridge human intent with robotic execution.

## Cognitive Planning Architecture

### Planning Pipeline Components

The cognitive planning pipeline consists of several key components:

1. **Goal Understanding**: Interpreting natural language commands
2. **Task Decomposition**: Breaking high-level goals into sub-tasks
3. **Action Sequencing**: Ordering actions logically and safely
4. **Context Awareness**: Incorporating environmental and state information
5. **Execution Planning**: Generating ROS 2 action sequences

### Sequential Action Planning

```python
class ActionSequencer:
    def __init__(self, robot_capabilities: Dict):
        self.capabilities = robot_capabilities
        self.action_precedence = {
            "pickup_object": ["navigate", "identify_object"],
            "place_object": ["navigate"],
            "open_door": ["navigate"],
            "press_button": ["navigate"]
        }

    def sequence_actions(self, decomposed_tasks: List[Dict]) -> List[Dict]:
        """
        Sequence actions logically and safely
        """
        # Validate action dependencies
        validated_sequence = []

        for task in decomposed_tasks:
            action = task["action"]

            # Check if required preceding actions are in sequence
            required_actions = self.action_precedence.get(action, [])
            for req_action in required_actions:
                if not self.is_action_available(req_action, validated_sequence):
                    # Add required action if not already present
                    required_task = self.create_required_task(req_action, task)
                    validated_sequence.append(required_task)

            validated_sequence.append(task)

        return validated_sequence

    def is_action_available(self, action: str, sequence: List[Dict]) -> bool:
        """
        Check if an action is available in the sequence
        """
        for task in sequence:
            if task["action"] == action:
                return True
        return False

    def create_required_task(self, action: str, dependent_task: Dict) -> Dict:
        """
        Create a required task based on dependencies
        """
        # This would use context to create appropriate parameters
        return {
            "action": action,
            "parameters": {},
            "required_by": dependent_task["action"]
        }

    def validate_plan(self, action_sequence: List[Dict]) -> Dict[str, Any]:
        """
        Validate the action sequence for feasibility
        """
        validation_result = {
            "valid": True,
            "errors": [],
            "warnings": [],
            "safety_issues": []
        }

        # Check for capability constraints
        for i, action in enumerate(action_sequence):
            action_type = action["action"]

            if action_type not in self.capabilities:
                validation_result["errors"].append(
                    f"Action {action_type} not supported by robot"
                )
                validation_result["valid"] = False

            # Check for safety constraints
            if self.is_unsafe_action(action, action_sequence[:i]):
                validation_result["safety_issues"].append(
                    f"Unsafe action: {action_type}"
                )

        return validation_result

    def is_unsafe_action(self, action: Dict, previous_actions: List[Dict]) -> bool:
        """
        Check if an action is unsafe given previous actions
        """
        # Example safety checks
        if action["action"] == "pickup_object":
            # Check if robot is already carrying something
            for prev_action in reversed(previous_actions):
                if prev_action["action"] == "pickup_object" and not prev_action.get("placed", False):
                    return True  # Can't carry multiple objects
        return False
```

```python
class ContextAwarePlanner:
    def __init__(self, node):
        self.node = node
        self.environment_context = {}
        self.object_locations = {}
        self.robot_state = {}

    def update_environment_context(self, map_data: Dict, object_data: Dict, robot_pose: Pose):
        """
        Update environmental context for planning
        """
        self.environment_context = map_data
        self.object_locations = object_data
        self.robot_state['pose'] = robot_pose

    def contextualize_plan(self, basic_plan: List[Dict]) -> List[Dict]:
        """
        Add environmental context to basic plan
        """
        contextualized_plan = []

        for step in basic_plan:
            contextualized_step = step.copy()

            # Add location context for navigation
            if step["action"] == "navigate":
                location = step["parameters"].get("location")
                if location in self.environment_context:
                    contextualized_step["parameters"]["waypoints"] = \
                        self.environment_context[location].get("waypoints", [])
                    contextualized_step["parameters"]["obstacles"] = \
                        self.environment_context[location].get("obstacles", [])

            # Add object context for manipulation
            elif step["action"] in ["pickup_object", "identify_object"]:
                target = step["parameters"].get("target")
                if target in self.object_locations:
                    contextualized_step["parameters"]["position"] = \
                        self.object_locations[target]["position"]
                    contextualized_step["parameters"]["properties"] = \
                        self.object_locations[target]["properties"]

            contextualized_plan.append(contextualized_step)

        return contextualized_plan

    def adapt_plan_to_context(self, plan: List[Dict], current_context: Dict) -> List[Dict]:
        """
        Adapt plan based on current environmental context
        """
        adapted_plan = []

        for step in plan:
            # Check if environment has changed
            if self.environment_has_changed(step, current_context):
                # Regenerate this step with new context
                adapted_step = self.regenerate_step_with_context(step, current_context)
                adapted_plan.append(adapted_step)
            else:
                adapted_plan.append(step)

        return adapted_plan

    def environment_has_changed(self, step: Dict, current_context: Dict) -> bool:
        """
        Check if environment has changed since plan creation
        """
        if step["action"] == "navigate":
            location = step["parameters"].get("location")
            if location in current_context.get("dynamic_obstacles", {}):
                return True
        return False

    def regenerate_step_with_context(self, step: Dict, current_context: Dict) -> Dict:
        """
        Regenerate a step with updated context
        """
        # Use LLM to regenerate step with new context
        prompt = f"""
        Original step: {json.dumps(step)}
        Current context: {json.dumps(current_context)}

        Regenerate this step considering the current environmental context.
        """

        try:
            response = self.node.llm_client.chat.completions.create(
                model="gpt-4-turbo",
                messages=[
                    {"role": "system", "content": "Regenerate the action step with current context."},
                    {"role": "user", "content": prompt}
                ],
                response_format={"type": "json_object"}
            )

            return json.loads(response.choices[0].message.content)

        except Exception as e:
            self.node.get_logger().error(f'Context regeneration error: {e}')
            return step  # Return original if regeneration fails
```

```python
class DynamicPlanUpdater:
    def __init__(self, node):
        self.node = node
        self.current_plan = []
        self.executed_steps = []
        self.failed_steps = []

    def update_plan_during_execution(self, current_state: Dict, feedback: Dict) -> List[Dict]:
        """
        Update plan based on execution feedback
        """
        if feedback.get("error"):
            # Handle step failure
            return self.handle_step_failure(current_state, feedback)
        elif feedback.get("deviation"):
            # Handle unexpected deviation
            return self.handle_deviation(current_state, feedback)
        else:
            # Continue with current plan
            return self.current_plan

    def handle_step_failure(self, current_state: Dict, feedback: Dict) -> List[Dict]:
        """
        Handle failure of current step
        """
        failed_step = self.current_plan[len(self.executed_steps)]
        self.failed_steps.append(failed_step)

        # Try alternative approach
        alternative_plan = self.generate_alternative_plan(
            failed_step, current_state, len(self.executed_steps)
        )

        if alternative_plan:
            # Insert alternative plan into remaining plan
            remaining_plan = self.current_plan[len(self.executed_steps) + 1:]
            new_plan = self.executed_steps + alternative_plan + remaining_plan
            self.current_plan = new_plan
            return new_plan
        else:
            # If no alternative, skip the failed step if possible
            remaining_plan = self.current_plan[len(self.executed_steps) + 1:]
            new_plan = self.executed_steps + remaining_plan
            self.current_plan = new_plan
            return new_plan

    def generate_alternative_plan(self, failed_step: Dict, current_state: Dict, step_index: int) -> List[Dict]:
        """
        Generate alternative plan for failed step
        """
        # Use LLM to suggest alternative
        prompt = f"""
        Failed step: {json.dumps(failed_step)}
        Current state: {json.dumps(current_state)}
        Step index: {step_index}

        Generate an alternative approach for the failed step.
        Consider the current state and what has already been executed.
        """

        try:
            response = self.node.llm_client.chat.completions.create(
                model="gpt-4-turbo",
                messages=[
                    {"role": "system", "content": "Generate alternative plan steps."},
                    {"role": "user", "content": prompt}
                ],
                response_format={"type": "json_object"}
            )

            result = json.loads(response.choices[0].message.content)
            return result.get("alternative_steps", [])

        except Exception as e:
            self.node.get_logger().error(f'Alternative plan generation error: {e}')
            return []

    def handle_deviation(self, current_state: Dict, feedback: Dict) -> List[Dict]:
        """
        Handle unexpected deviation from plan

        if deviation_type == "object_not_found":
            # Look for object in nearby locations

        elif deviation_type == "path_blocked":

            new_plan = self.recalculate_path(current_state, feedback)
        else:
            # Continue with plan adjustment
            new_plan = self.adjust_plan_for_deviation(current_state, feedback)

        return new_plan
```

```python
        plan = self.generate_cognitive_plan(command, context)

        # Cache the result
        if len(self.plan_cache) >= self.max_cache_size:
            # Remove oldest entry
            oldest_key = next(iter(self.plan_cache))
            del self.plan_cache[oldest_key]

        self.plan_cache[cache_key] = plan
        return plan

    def precompute_common_plans(self):
        """
        Precompute plans for common commands
        """
        common_commands = [
            "go to kitchen",
            "pick up the red cup",
            "bring me water",
            "clean the table"
        ]

        base_context = self.get_base_context()

        for command in common_commands:
            try:
                plan = self.generate_cognitive_plan(command, base_context)
                cache_key = self.get_command_context_hash(command, base_context)
                self.plan_cache[cache_key] = plan
                self.node.get_logger().info(f'Precomputed plan for: {command}')
            except Exception as e:
                self.node.get_logger().error(f'Precomputation error for {command}: {e}')
```
