# Vision for Object Understanding

This chapter covers integrating camera-based object detection and localization to connect perception results to LLM-generated plans. We'll explore how to enable robots to see, understand, and interact with objects in their environment.

## Introduction to Robot Vision

Robot vision is a critical component of the VLA system, enabling robots to perceive and understand their environment. Through cameras and computer vision algorithms, robots can detect objects, understand spatial relationships, and make informed decisions about how to interact with the world.

## Object Detection Fundamentals

Object detection in robotics typically involves:

1. **Image Acquisition**: Capturing images from one or more cameras
2. **Preprocessing**: Enhancing images for better detection
3. **Detection**: Identifying objects and their locations
4. **Classification**: Determining what each object is
5. **Localization**: Estimating 3D positions from 2D detections

```python
import cv2
import numpy as np
import torch
from torchvision import transforms

class ObjectDetector:
    def __init__(self, model_path="yolov8n.pt"):
        # Load pre-trained object detection model
        self.model = torch.hub.load('ultralytics/yolov5', 'custom', path=model_path)
        self.transform = transforms.Compose([
            transforms.ToTensor(),
        ])

    def detect_objects(self, image):
        """Detect objects in an image"""
        # Perform inference
        results = self.model(image)

        # Parse results
        detections = []
        for detection in results.xyxy[0]:  # x1, y1, x2, y2, confidence, class
            x1, y1, x2, y2, conf, cls = detection
            detections.append({
                "class_id": int(cls),
                "class_name": self.model.names[int(cls)],
                "confidence": float(conf),
                "bbox": {
                    "x": float(x1),
                    "y": float(y1),
                    "width": float(x2 - x1),
                    "height": float(y2 - y1)
                },
                "center": {
                    "x": float((x1 + x2) / 2),
                    "y": float((y1 + y2) / 2)
                }
            })

        return detections

    def filter_detections(self, detections, confidence_threshold=0.5, classes_of_interest=None):
        """Filter detections based on confidence and class"""
        filtered = []
        for detection in detections:
            if detection["confidence"] < confidence_threshold:
                continue

            if classes_of_interest and detection["class_name"] not in classes_of_interest:
                continue

            filtered.append(detection)

        return filtered
```

## Scene Understanding and Description

For LLM integration, we need to convert visual information into natural language descriptions:

```python
class SceneDescriber:
    def __init__(self):
        self.object_relationships = {
            "on": ["plate", "cup", "bottle"],
            "in": ["box", "drawer", "container"],
            "next_to": ["chair", "table", "sofa"]
        }

    def describe_scene(self, detections, camera_pose):
        """Generate a natural language description of the scene"""
        if not detections:
            return "The scene appears to be empty - no objects detected."

        # Group objects by location
        objects_by_location = self._group_objects_by_location(detections, camera_pose)

        # Generate description
        description_parts = []

        # Describe each location
        for location, objects in objects_by_location.items():
            if objects:
                object_names = [obj["class_name"] for obj in objects]
                count = len(objects)

                if count == 1:
                    description_parts.append(f"There is a {object_names[0]} {location}.")
                else:
                    description_parts.append(f"There are {count} objects {location}: {', '.join(object_names)}.")

        # Add spatial relationships
        spatial_relationships = self._describe_spatial_relationships(detections)
        if spatial_relationships:
            description_parts.extend(spatial_relationships)

        return " ".join(description_parts)

    def _group_objects_by_location(self, detections, camera_pose):
        """Group objects by their location in the scene"""
        # Simplified grouping by rough spatial location
        locations = {
            "in front of the camera": [],
            "to the left": [],
            "to the right": [],
            "far away": [],
            "close by": []
        }

        for detection in detections:
            # Determine location based on 2D position and depth
            center_x = detection["center"]["x"]
            distance = detection.get("distance", 1.0)  # Assume distance is available

            if distance < 0.5:
                location = "close by"
            elif distance > 2.0:
                location = "far away"
            elif center_x < 200:  # Assuming image width around 640
                location = "to the left"
            elif center_x > 440:
                location = "to the right"
            else:
                location = "in front of the camera"

            locations[location].append(detection)

        return locations

    def _describe_spatial_relationships(self, detections):
        """Describe spatial relationships between objects"""
        relationships = []

        # Find pairs of objects that have spatial relationships
        for i, obj1 in enumerate(detections):
            for j, obj2 in enumerate(detections[i+1:]):
                # Calculate distance between objects
                dist = self._calculate_object_distance(obj1, obj2)

                if dist < 0.3:  # Objects are close
                    relationships.append(f"The {obj1['class_name']} is close to the {obj2['class_name']}.")

        return relationships

    def _calculate_object_distance(self, obj1, obj2):
        """Calculate distance between two objects"""
        # Simplified calculation - in a real system, this would use 3D positions
        pos1 = obj1.get("position", {"x": 0, "y": 0, "z": 0})
        pos2 = obj2.get("position", {"x": 0, "y": 0, "z": 0})

        dx = pos1["x"] - pos2["x"]
        dy = pos1["y"] - pos2["y"]
        dz = pos1["z"] - pos2["z"]

        return (dx**2 + dy**2 + dz**2)**0.5
```

## Integration with LLM Planning

The vision system integrates with LLM planning by providing object information that can be used to refine plans:

    def __init__(self):
        self.object_detector = ObjectDetector()
        self.scene_describer = SceneDescriber()

    async def process_vision_for_planning(self, image, plan_request):
        """Process vision data to enhance LLM planning"""
        # Detect objects in the image
        detections = self.object_detector.detect_objects(image)

        # Filter for objects relevant to the plan request
        relevant_classes = self._get_relevant_classes(plan_request)
        filtered_detections = self.object_detector.filter_detections(
            detections,
            confidence_threshold=0.6,
            classes_of_interest=relevant_classes
        )

        # Localize objects in 3D space (if depth information available)
        localized_objects = await self._localize_objects_3d(filtered_detections)

        # Create vision-enhanced context for LLM
        vision_context = {
            "detected_objects": localized_objects,
            "scene_description": scene_description,
            "object_locations": self._create_object_location_map(localized_objects),
            "spatial_relationships": self._extract_spatial_relationships(localized_objects),
            "timestamp": time.time()
        }

        return vision_context

    def _get_relevant_classes(self, plan_request):
        """Get object classes relevant to the current plan request"""
        # Extract object types from the plan request
        goal = plan_request.get("goal", "").lower()
        relevant_classes = []

        # Simple keyword matching (in practice, this would be more sophisticated)
        if "cup" in goal or "drink" in goal:
            relevant_classes.extend(["cup", "bottle", "glass"])
        if "book" in goal:
            relevant_classes.append("book")
        if "chair" in goal:
            relevant_classes.append("chair")
        if "table" in goal:
            relevant_classes.append("table")
        if "box" in goal:
            relevant_classes.append("box")

        return relevant_classes

    async def _localize_objects_3d(self, detections):
        """Localize detected objects in 3D space"""
        # In a real implementation, this would use depth information
        # For simulation, we'll add placeholder 3D positions
        localized = []
        for detection in detections:
            detection_3d = detection.copy()
            # Add placeholder 3D position (in a real system, this would come from depth data)
            detection_3d["position"] = {
                "x": float(np.random.uniform(-1, 1)),
                "y": float(np.random.uniform(-1, 1)),
                "z": float(np.random.uniform(0.5, 2.0))
            }
            detection_3d["distance"] = (detection_3d["position"]["x"]**2 +
                                       detection_3d["position"]["y"]**2 +
                                       detection_3d["position"]["z"]**2)**0.5
            localized.append(detection_3d)

        return localized

    def _create_object_location_map(self, localized_objects):
        """Create a map of object names to their locations"""
        location_map = {}
        for obj in localized_objects:
            name = obj["class_name"]
            if name not in location_map:
                location_map[name] = []
            location_map[name].append(obj["position"])
        return location_map

    def _extract_spatial_relationships(self, localized_objects):
        """Extract spatial relationships between objects"""
        relationships = []
        for i, obj1 in enumerate(localized_objects):
            for j, obj2 in enumerate(localized_objects[i+1:]):
                distance = self._calculate_3d_distance(obj1["position"], obj2["position"])
                if distance < 0.5:  # Objects are close
                    relationships.append({
                        "object1": obj1["class_name"],
                        "object2": obj2["class_name"],
                        "relationship": "near",
                        "distance": distance
                    })
        return relationships

    def _calculate_3d_distance(self, pos1, pos2):
        """Calculate 3D distance between two positions"""
        dx = pos1["x"] - pos2["x"]
        dy = pos1["y"] - pos2["y"]
        dz = pos1["z"] - pos2["z"]
        return (dx**2 + dy**2 + dz**2)**0.5

## Real-time Vision Processing

For interactive applications, vision processing needs to be efficient:

```python
import asyncio
import threading
from queue import Queue

class RealTimeVisionProcessor:
    def __init__(self, fps_target=10):
        self.fps_target = fps_target
        self.frame_queue = Queue(maxsize=2)  # Limit queue size
        self.result_queue = Queue()
        self.running = False
        self.vision_thread = None

    def start_processing(self):
        """Start real-time vision processing"""
        self.running = True
        self.vision_thread = threading.Thread(target=self._vision_worker)
        self.vision_thread.start()

    def stop_processing(self):
        """Stop real-time vision processing"""
        self.running = False
        if self.vision_thread:
            self.vision_thread.join()

    def submit_frame(self, frame):
        """Submit a frame for processing"""
        try:
            self.frame_queue.put_nowait(frame)
        except:
            # Queue is full, drop the frame
            pass

    def get_latest_results(self):
        """Get the most recent vision processing results"""
        results = []
        try:
            while True:
                results.append(self.result_queue.get_nowait())
        except:
            pass  # No more results in queue

        # Return the most recent result
        return results[-1] if results else None

    def _vision_worker(self):
        """Background worker for vision processing"""
        detector = ObjectDetector()

        while self.running:
            try:
                # Get frame from queue
                frame = self.frame_queue.get(timeout=1.0)

                # Process frame
                detections = detector.detect_objects(frame)

                # Put results in output queue
                result = {
                    "detections": detections,
                    "timestamp": time.time()
                }
                self.result_queue.put(result)

                # Control frame rate
                time.sleep(1.0 / self.fps_target)

            except:
                # Timeout or other error, continue loop
                continue
```
