# Vision for Object Understanding

## Introduction to Robot Vision

Computer vision is fundamental to robotic perception, enabling robots to understand and interact with their environment. In Vision-Language-Action (VLA) systems, vision provides the crucial link between language understanding and physical action execution. This chapter covers the essential concepts and implementation of vision-based object understanding for humanoid robots.

## Vision System Architecture

### Multi-Modal Perception Pipeline

The vision system in VLA applications typically follows this pipeline:

1. **Image Acquisition**: Capturing images from cameras
2. **Preprocessing**: Image enhancement and normalization
3. **Feature Extraction**: Extracting relevant visual features
4. **Object Detection**: Identifying and localizing objects
5. **Object Recognition**: Classifying identified objects
6. **Pose Estimation**: Determining object position and orientation
7. **Scene Understanding**: Interpreting object relationships

```python
class AdvancedObjectDetector:
    def __init__(self, node):
        self.node = node
        self.detection_models = {}
        self.load_detection_models()

    def load_detection_models(self):
        """
        Load multiple detection models for different scenarios
        """
        try:
            # General object detection
            self.detection_models['general'] = torch.hub.load(
                'ultralytics/yolov5', 'yolov5m', pretrained=True
            )

            # Custom model for specific objects (if available)
            # self.detection_models['custom'] = self.load_custom_model()

        except Exception as e:
            self.node.get_logger().error(f'Failed to load detection models: {e}')

    def detect_objects_multiscale(self, image):
        """
        Perform multi-scale object detection
        """
        # Different scales for better detection
        scales = [0.5, 1.0, 1.5]
        all_detections = []

        for scale in scales:
            scaled_image = cv2.resize(image, None, fx=scale, fy=scale)
            detections = self.perform_single_scale_detection(scaled_image)
            # Adjust coordinates back to original scale
            for det in detections:
                det['bbox'] = [x/scale for x in det['bbox']]
            all_detections.extend(detections)

        # Merge overlapping detections
        merged_detections = self.merge_overlapping_detections(all_detections)
        return merged_detections

    def perform_single_scale_detection(self, image):
        """
        Perform detection on single scale image
        """
        image_rgb = cv2.cvtColor(image, cv2.COLOR_BGR2RGB)
        input_tensor = T.ToTensor()(image_rgb).unsqueeze(0)

        with torch.no_grad():
            results = self.detection_models['general'](input_tensor)

        detections = []
        for i, det in enumerate(results.xyxy):
            for *xyxy, conf, cls in det:
                if conf > 0.5:  # Confidence threshold
                    x1, y1, x2, y2 = map(int, xyxy)
                    detections.append({
                        'bbox': [x1, y1, x2, y2],
                        'confidence': float(conf),
                        'class_id': int(cls),
                        'class_name': self.get_class_name(int(cls))
                    })

        return detections

    def merge_overlapping_detections(self, detections):
        """
        Merge overlapping detections using non-maximum suppression
        """
        if not detections:
            return []

        # Convert to format for NMS
        boxes = np.array([det['bbox'] for det in detections])
        scores = np.array([det['confidence'] for det in detections])

        # Apply NMS
        keep_indices = self.non_max_suppression(boxes, scores, iou_threshold=0.5)

        return [detections[i] for i in keep_indices]

    def non_max_suppression(self, boxes, scores, iou_threshold=0.5):
        """
        Perform non-maximum suppression
        """
        # Sort by scores
        indices = np.argsort(scores)[::-1]

        keep = []
        while len(indices) > 0:
            current = indices[0]
            keep.append(current)

            if len(indices) == 1:
                break

            # Calculate IoU with remaining boxes
            remaining = indices[1:]
            ious = self.calculate_iou(boxes[current], boxes[remaining])

            # Keep boxes with IoU below threshold
            indices = remaining[ious < iou_threshold]

        return keep

    def calculate_iou(self, box1, boxes):
        """
        Calculate IoU between one box and multiple boxes
        """
        # Calculate intersection
        x1 = np.maximum(box1[0], boxes[:, 0])
        y1 = np.maximum(box1[1], boxes[:, 1])
        x2 = np.minimum(box1[2], boxes[:, 2])
        y2 = np.minimum(box1[3], boxes[:, 3])

        intersection = np.maximum(0, x2 - x1) * np.maximum(0, y2 - y1)
        area1 = (box1[2] - box1[0]) * (box1[3] - box1[1])
        area2 = (boxes[:, 2] - boxes[:, 0]) * (boxes[:, 3] - boxes[:, 1])

        union = area1 + area2 - intersection
        iou = intersection / union
        return iou
```

## 3D Object Pose Estimation

### Pose Estimation from RGB-D Data

```python
class PoseEstimator:
    def __init__(self, node):
        self.node = node
        self.camera_matrix = None
        self.dist_coeffs = None

    def set_camera_params(self, camera_info):
        """
        Set camera intrinsic parameters
        """
        self.camera_matrix = np.array(camera_info.k).reshape(3, 3)
        self.dist_coeffs = np.array(camera_info.d)

    def estimate_object_pose(self, object_mask, object_class, depth_image):
        """
        Estimate 3D pose of object from mask and depth data
        """
        # Find contours of object mask
        contours, _ = cv2.findContours(
            object_mask.astype(np.uint8),
            cv2.RETR_EXTERNAL,
            cv2.CHAIN_APPROX_SIMPLE
        )

        if not contours:
            return None

        # Get largest contour
        largest_contour = max(contours, key=cv2.contourArea)

        # Get 2D points from contour
        points_2d = largest_contour.reshape(-1, 2).astype(np.float32)

        # Get corresponding 3D points from depth image
        points_3d = self.depth_to_3d_points(points_2d, depth_image)

        if len(points_3d) < 4:
            return None

        # Estimate pose using PnP
        success, rvec, tvec = cv2.solvePnP(
            points_3d, points_2d, self.camera_matrix, self.dist_coeffs
        )

        if success:
            # Convert rotation vector to rotation matrix
            rotation_matrix, _ = cv2.Rodrigues(rvec)

            # Create pose dictionary
            pose = {
                'translation': tvec.flatten().tolist(),
                'rotation_matrix': rotation_matrix.tolist(),
                'object_class': object_class
            }

            return pose
        else:
            return None

    def depth_to_3d_points(self, points_2d, depth_image):
        """
        Convert 2D points to 3D points using depth information
        """
        points_3d = []

        for point in points_2d:
            u, v = int(point[0]), int(point[1])

            if 0 <= u < depth_image.shape[1] and 0 <= v < depth_image.shape[0]:
                depth = depth_image[v, u]

                if depth > 0:  # Valid depth
                    # Convert to 3D using camera intrinsics
                    x = (u - self.camera_matrix[0, 2]) * depth / self.camera_matrix[0, 0]
                    y = (v - self.camera_matrix[1, 2]) * depth / self.camera_matrix[1, 1]
                    z = depth

                    points_3d.append([x, y, z])

        return np.array(points_3d, dtype=np.float32)

    def create_object_mask(self, image, detection_bbox):
        """
        Create segmentation mask for detected object
        """
        mask = np.zeros(image.shape[:2], dtype=np.uint8)
        x1, y1, x2, y2 = detection_bbox

        # Simple bounding box mask
        # In practice, you might use instance segmentation
        mask[y1:y2, x1:x2] = 255

        return mask
```

```python
class FeatureExtractor:
    def __init__(self, node):
        self.node = node
        self.feature_model = self.load_feature_model()

    def load_feature_model(self):
        """
        Load pre-trained feature extraction model
        """
        try:
            # Using ResNet50 as feature extractor
            model = torch.hub.load('pytorch/vision:v0.10.0', 'resnet50', pretrained=True)

            # Remove classification layer to get feature embeddings
            feature_layers = list(model.children())[:-1]
            feature_extractor = torch.nn.Sequential(*feature_layers)

            feature_extractor.eval()
            return feature_extractor
        except Exception as e:
            self.node.get_logger().error(f'Failed to load feature model: {e}')
            return None

    def extract_object_features(self, object_image):
        """
        Extract deep features from object image
        """
        if self.feature_model is None:
            return None

        # Preprocess image
        preprocess = T.Compose([
            T.ToPILImage() if not isinstance(object_image, Image.Image) else lambda x: x,
            T.Resize((224, 224)),
            T.ToTensor(),
            T.Normalize(mean=[0.485, 0.456, 0.406], std=[0.229, 0.224, 0.225])
        ])

        input_tensor = preprocess(object_image).unsqueeze(0)

        with torch.no_grad():
            if torch.cuda.is_available():
                input_tensor = input_tensor.cuda()
                self.feature_model = self.feature_model.cuda()

            features = self.feature_model(input_tensor)
            features = features.squeeze().cpu().numpy()

        return features

    def extract_spatial_features(self, image, bbox):
        """
        Extract spatial features from object region
        """
        x1, y1, x2, y2 = bbox
        object_region = image[y1:y2, x1:x2]

        # Extract various spatial features
        features = {}

        # Color histogram
        hist_b = cv2.calcHist([object_region], [0], None, [32], [0, 256])
        hist_g = cv2.calcHist([object_region], [1], None, [32], [0, 256])
        hist_r = cv2.calcHist([object_region], [2], None, [32], [0, 256])

        features['color_histogram'] = np.concatenate([hist_b.flatten(), hist_g.flatten(), hist_r.flatten()])

        # Texture features (using Local Binary Patterns)
        gray = cv2.cvtColor(object_region, cv2.COLOR_BGR2GRAY)
        lbp = self.calculate_lbp(gray)
        features['texture_features'] = lbp

        # Shape features
        features['area'] = (x2 - x1) * (y2 - y1)
        features['aspect_ratio'] = (x2 - x1) / (y2 - y1) if (y2 - y1) != 0 else 0
        features['compactness'] = 4 * np.pi * features['area'] / (cv2.contourArea(np.array([[[x1,y1], [x2,y1], [x2,y2], [x1,y2]]])))**2

        return features

    def calculate_lbp(self, image):
        """
        Calculate Local Binary Pattern features
        """
        # Simplified LBP calculation
        # In practice, use scikit-image or other optimized libraries
        lbp_hist, _ = np.histogram(image.flatten(), bins=256)
        return lbp_hist
```

```python
class SceneUnderstanding:
    def __init__(self, node):
        self.node = node
        self.object_contexts = {
            'kitchen': ['cup', 'plate', 'fork', 'knife', 'spoon', 'bottle', 'refrigerator'],
            'living_room': ['couch', 'tv', 'remote', 'book', 'vase'],
            'bedroom': ['bed', 'pillow', 'clock', 'lamp'],
            'office': ['laptop', 'keyboard', 'mouse', 'book', 'pen']
        }

    def analyze_scene_context(self, detected_objects, image_shape):
        """
        Analyze scene context based on object relationships
        """
        scene_analysis = {
            'likely_room_type': self.infer_room_type(detected_objects),
            'object_relationships': self.find_object_relationships(detected_objects),
            'interaction_potential': self.assess_interaction_potential(detected_objects),
            'spatial_layout': self.analyze_spatial_layout(detected_objects, image_shape)
        }

        return scene_analysis

    def infer_room_type(self, detected_objects):
        """
        Infer room type based on detected objects
        """
        object_names = [obj['class_name'] for obj in detected_objects]
        room_scores = {}

        for room_type, expected_objects in self.object_contexts.items():
            score = sum(1 for obj in object_names if obj in expected_objects)
            room_scores[room_type] = score

        # Return room type with highest score
        return max(room_scores, key=room_scores.get) if room_scores else 'unknown'

    def find_object_relationships(self, detected_objects):
        """
        Find spatial and functional relationships between objects
        """
        relationships = []

        for i, obj1 in enumerate(detected_objects):
            for j, obj2 in enumerate(detected_objects[i+1:], i+1):
                relationship = self.calculate_relationship(obj1, obj2)
                if relationship:
                    relationships.append(relationship)

        return relationships

    def calculate_relationship(self, obj1, obj2):
        """
        Calculate relationship between two objects
        """
        bbox1 = obj1['bbox']
        bbox2 = obj2['bbox']

        # Calculate spatial relationship
        center1 = ((bbox1[0] + bbox1[2]) / 2, (bbox1[1] + bbox1[3]) / 2)
        center2 = ((bbox2[0] + bbox2[2]) / 2, (bbox2[1] + bbox2[3]) / 2)

        distance = np.sqrt((center1[0] - center2[0])**2 + (center1[1] - center2[1])**2)

        # Calculate if objects are adjacent (simple heuristic)
        avg_size = ((bbox1[2] - bbox1[0]) + (bbox1[3] - bbox1[1]) +
                    (bbox2[2] - bbox2[0]) + (bbox2[3] - bbox2[1])) / 4
        is_adjacent = distance < avg_size * 2

        relationship = {
            'objects': [obj1['class_name'], obj2['class_name']],
            'distance': distance,
            'is_adjacent': is_adjacent,
            'spatial_relationship': self.get_spatial_relationship(bbox1, bbox2)
        }

        return relationship

    def get_spatial_relationship(self, bbox1, bbox2):
        """
        Determine spatial relationship between two bounding boxes
        """
        center1 = ((bbox1[0] + bbox1[2]) / 2, (bbox1[1] + bbox1[3]) / 2)
        center2 = ((bbox2[0] + bbox2[2]) / 2, (bbox2[1] + bbox2[3]) / 2)

        dx = center2[0] - center1[0]
        dy = center2[1] - center1[1]

        # Determine directional relationship
        if abs(dx) > abs(dy):
            direction = 'left' if dx < 0 else 'right'
        else:
            direction = 'above' if dy < 0 else 'below'

        return direction

    def assess_interaction_potential(self, detected_objects):
        """
        Assess potential for robot-object interactions
        """
        interaction_potential = []

        for obj in detected_objects:
            potential = {
                'object': obj['class_name'],
                'manipulable': self.is_manipulable(obj),
                'approachable': self.is_approachable(obj),
                'graspable': self.is_graspable(obj)
            }
            interaction_potential.append(potential)

        return interaction_potential

    def is_manipulable(self, obj):
        """
        Determine if object is manipulable by robot
        """
        # Simple heuristic based on object class
        manipulable_classes = [
            'cup', 'bottle', 'book', 'laptop', 'box', 'bowl', 'plate',
            'fork', 'knife', 'spoon', 'pen', 'remote', 'phone'
        ]
        return obj['class_name'] in manipulable_classes

    def is_approachable(self, obj):
        """
        Determine if object is approachable
        """
        # This would involve path planning and accessibility analysis
        # For now, assume all objects are approachable
        return True

    def is_graspable(self, obj):
        """
        Determine if object is graspable
        """
        bbox = obj['bbox']
        width = bbox[2] - bbox[0]
        height = bbox[3] - bbox[1]

        # Simple size-based heuristic
        min_size = 10  # pixels
        max_size = 500  # pixels
        return min_size < width < max_size and min_size < height < max_size

    def analyze_spatial_layout(self, detected_objects, image_shape):
        """
        Analyze spatial layout of scene
        """
        layout = {
            'object_density': self.calculate_object_density(detected_objects, image_shape),
            'spatial_distribution': self.analyze_spatial_distribution(detected_objects),
            'focal_points': self.find_focal_points(detected_objects)
        }

        return layout

    def calculate_object_density(self, detected_objects, image_shape):
        """
        Calculate object density in the scene
        """
        image_area = image_shape[0] * image_shape[1]
        covered_area = sum((obj['bbox'][2] - obj['bbox'][0]) * (obj['bbox'][3] - obj['bbox'][1])
                          for obj in detected_objects)
        density = covered_area / image_area if image_area > 0 else 0
        return density

    def analyze_spatial_distribution(self, detected_objects):
        """
        Analyze how objects are distributed in space
        """
        if not detected_objects:
            return 'empty'

        centers = [(obj['bbox'][0] + obj['bbox'][2]) / 2 for obj in detected_objects]
        x_variance = np.var(centers)

        if x_variance < 1000:  # Low variance = clustered
            return 'clustered'
        else:
            return 'distributed'
```

```python
                found_objects.append(keyword)

        return found_objects

    def get_canonical_name(self, name):
        """

        return self.object_vocabulary.get(name.lower(), name.lower())

    def is_match(self, target, detected_class):
        """

        """

        return target in detected_class or detected_class in target

    def get_spatial_info(self, obj):
        """
        Get spatial information about object
        """
        bbox = obj['bbox']
        center_x = (bbox[0] + bbox[2]) / 2
        center_y = (bbox[1] + bbox[3]) / 2

        # Relative position in image (left, center, right / top, center, bottom)
        h_pos = 'left' if center_x < 213 else 'center' if center_x < 426 else 'right'
        v_pos = 'top' if center_y < 160 else 'center' if center_y < 320 else 'bottom'

        return {
            'position': f'{v_pos} {h_pos}',
            'bbox': bbox,
            'center': [center_x, center_y]
        }

            'spatial_layout': scene_analysis.get('spatial_layout', {})
```

### Efficient Processing Pipeline

```python
import threading
import queue
from collections import deque

class EfficientVisionPipeline:
    def __init__(self, node):
        self.node = node
        self.input_queue = queue.Queue(maxsize=2)  # Limit queue size
        self.result_queue = queue.Queue(maxsize=2)

        # Processing threads
        self.processing_thread = threading.Thread(target=self.processing_loop)
        self.processing_thread.daemon = True
        self.processing_thread.start()

        # Frame skipping for real-time performance
        self.frame_counter = 0
        self.process_every_n_frames = 3  # Process every 3rd frame

        # Object tracking to maintain consistency
        self.tracked_objects = {}
        self.next_object_id = 0

    def process_image_efficiently(self, image, header):
        """
        Process image with efficiency optimizations
        """
        self.frame_counter += 1

        # Skip frames for performance
        if self.frame_counter % self.process_every_n_frames != 0:
            return

        try:
            # Add to processing queue (non-blocking)
            if self.input_queue.qsize() < 2:  # Don't overwhelm queue
                self.input_queue.put((image, header), block=False)
        except queue.Full:
            self.node.get_logger().warn('Processing queue full, dropping frame')

    def processing_loop(self):
        """
        Background processing loop
        """
        while True:
            try:
                image, header = self.input_queue.get(timeout=1.0)

                # Perform vision processing
                results = self.perform_vision_processing(image)

                # Add results to output queue
                self.result_queue.put((results, header))

            except queue.Empty:
                continue  # Keep loop running
            except Exception as e:
                self.node.get_logger().error(f'Processing loop error: {e}')

    def perform_vision_processing(self, image):
        """
        Perform actual vision processing
        """
        # Object detection
        detections = self.perform_object_detection(image)

        # Object tracking
        tracked_detections = self.update_object_tracking(detections)

        # Scene analysis
        scene_analysis = self.analyze_scene_context(tracked_detections, image.shape)

        return {
            'detections': tracked_detections,
            'scene_analysis': scene_analysis,
            'timestamp': time.time()
        }

    def update_object_tracking(self, detections):
        """
        Update object tracking to maintain consistent IDs
        """
        # Simple tracking by bounding box overlap
        for detection in detections:
            bbox = detection['bbox']
            detection['object_id'] = self.assign_object_id(bbox)

        return detections

    def assign_object_id(self, bbox):
        """
        Assign consistent ID to object based on position
        """
        # Calculate center of bounding box
        center = ((bbox[0] + bbox[2]) / 2, (bbox[1] + bbox[3]) / 2)

        # Find existing object that matches this position
        best_match_id = None
        best_iou = 0

        for obj_id, obj_info in self.tracked_objects.items():
            existing_center = obj_info['center']
            distance = np.sqrt((center[0] - existing_center[0])**2 + (center[1] - existing_center[1])**2)

            if distance < 50:  # Threshold for considering same object
                iou = self.calculate_bbox_iou(bbox, obj_info['bbox'])
                if iou > best_iou:
                    best_iou = iou
                    best_match_id = obj_id

        if best_match_id is not None:
            # Update existing object
            self.tracked_objects[best_match_id].update({
                'center': center,
                'bbox': bbox
            })
            return best_match_id
        else:
            # Create new object
            new_id = self.next_object_id
            self.tracked_objects[new_id] = {
                'center': center,
                'bbox': bbox,
                'creation_time': time.time()
            }
            self.next_object_id += 1
            return new_id

    def calculate_bbox_iou(self, bbox1, bbox2):
        """
        Calculate Intersection over Union of two bounding boxes
        """
        x1 = max(bbox1[0], bbox2[0])
        y1 = max(bbox1[1], bbox2[1])
        x2 = min(bbox1[2], bbox2[2])
        y2 = min(bbox1[3], bbox2[3])

        intersection = max(0, x2 - x1) * max(0, y2 - y1)
        area1 = (bbox1[2] - bbox1[0]) * (bbox1[3] - bbox1[1])
        area2 = (bbox2[2] - bbox2[0]) * (bbox2[3] - bbox2[1])
        union = area1 + area2 - intersection

        return intersection / union if union > 0 else 0
```

```python
import sqlite3
import pickle
from datetime import datetime, timedelta

class ObjectMemory:
    def __init__(self, node, db_path="object_memory.db"):
        self.node = node
        self.db_path = db_path
        self.init_database()

    def init_database(self):
        """
        Initialize object memory database
        """
        conn = sqlite3.connect(self.db_path)
        cursor = conn.cursor()

        # Create objects table
        cursor.execute('''
            CREATE TABLE IF NOT EXISTS objects (
                id INTEGER PRIMARY KEY,
                name TEXT NOT NULL,
                class TEXT NOT NULL,
                features BLOB,
                location TEXT,
                last_seen TIMESTAMP,
                confidence REAL,
                interaction_count INTEGER DEFAULT 0
            )
        ''')

        conn.commit()
        conn.close()

    def remember_object(self, obj_info):
        """
        Store object information in memory
        """
        conn = sqlite3.connect(self.db_path)
        cursor = conn.cursor()

        # Serialize features
        features_blob = pickle.dumps(obj_info.get('features', {}))

        cursor.execute('''
            INSERT OR REPLACE INTO objects
            (name, class, features, location, last_seen, confidence, interaction_count)
            VALUES (?, ?, ?, ?, ?, ?, ?)
        ''', (
            obj_info['name'],
            obj_info['class'],
            features_blob,
            obj_info.get('location', ''),
            datetime.now(),
            obj_info.get('confidence', 1.0),
            obj_info.get('interaction_count', 0)
        ))

        conn.commit()
        conn.close()

    def recall_objects(self, class_filter=None, location_filter=None, hours_back=24):
        """
        Recall objects from memory
        """
        conn = sqlite3.connect(self.db_path)
        cursor = conn.cursor()

        query = "SELECT * FROM objects WHERE last_seen > ?"
        params = [datetime.now() - timedelta(hours=hours_back)]

        if class_filter:
            query += " AND class = ?"
            params.append(class_filter)

        if location_filter:
            query += " AND location LIKE ?"
            params.append(f'%{location_filter}%')

        query += " ORDER BY last_seen DESC"

        cursor.execute(query, params)
        rows = cursor.fetchall()

        objects = []
        for row in rows:
            obj = {
                'id': row[0],
                'name': row[1],
                'class': row[2],
                'features': pickle.loads(row[3]) if row[3] else {},
                'location': row[4],
                'last_seen': row[5],
                'confidence': row[6],
                'interaction_count': row[7]
            }
            objects.append(obj)

        conn.close()
        return objects

    def update_object_interaction(self, object_id, success=True):
        """
        Update object interaction statistics
        """
        conn = sqlite3.connect(self.db_path)
        cursor = conn.cursor()

        cursor.execute('''
            UPDATE objects SET interaction_count = interaction_count + 1
            WHERE id = ?
        ''', (object_id,))

        conn.commit()
        conn.close()
```
