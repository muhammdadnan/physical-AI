## VLA Architecture Patterns

### End-to-End Architectures
Vision-Language-Action models that process all modalities jointly:

```python
import torch
import torch.nn as nn

class EndToEndVLA(nn.Module):
    def __init__(self, vision_encoder, language_encoder, action_head):
        super().__init__()
        self.vision_encoder = vision_encoder
        self.language_encoder = language_encoder
        self.fusion_layer = nn.MultiheadAttention(
            embed_dim=512, num_heads=8)
        self.action_head = action_head

    def forward(self, images, text, proprioception=None):
        # Encode visual features
        visual_features = self.vision_encoder(images)

        # Encode language features
        lang_features = self.language_encoder(text)

        # Fuse modalities
        fused_features, _ = self.fusion_layer(
            visual_features, lang_features, lang_features)

        # Generate actions
        actions = self.action_head(fused_features)
        return actions
```

## Multimodal Fusion

### Early Fusion
Combining modalities at the input level:

```python
class EarlyFusion(nn.Module):
    def __init__(self):
        super().__init__()
        self.vision_proj = nn.Linear(512, 256)
        self.lang_proj = nn.Linear(512, 256)
        self.fusion = nn.Linear(512, 512)  # 256 + 256

    def forward(self, vision_features, lang_features):
        v_proj = self.vision_proj(vision_features)
        l_proj = self.lang_proj(lang_features)

        # Concatenate and fuse
        combined = torch.cat([v_proj, l_proj], dim=-1)
        fused = self.fusion(combined)
        return fused
```

### Late Fusion
Combining outputs from separate modalities:

```python
class LateFusion(nn.Module):
    def __init__(self):
        super().__init__()
        self.vision_head = nn.Linear(512, 256)
        self.lang_head = nn.Linear(512, 256)
        self.action_predictor = nn.Linear(512, 7)  # 7-DoF robot arm

    def forward(self, vision_features, lang_features):
        v_out = self.vision_head(vision_features)
        l_out = self.lang_head(lang_features)

        # Combine outputs
        combined = torch.cat([v_out, l_out], dim=-1)
        actions = self.action_predictor(combined)
        return actions
```

### Cross-Attention Fusion
Using attention mechanisms to combine modalities:

```python
class CrossAttentionFusion(nn.Module):
    def __init__(self):
        super().__init__()
        self.cross_attn = nn.MultiheadAttention(
            embed_dim=512, num_heads=8, batch_first=True)
        self.norm = nn.LayerNorm(512)

    def forward(self, vision_features, lang_features):
        # Vision attends to language
        attended_vision, _ = self.cross_attn(
            vision_features, lang_features, lang_features)

        # Normalize and return
        return self.norm(attended_vision + vision_features)
```

### Action Generation Transformers
```python
class ActionTransformer(nn.Module):
    def __init__(self, d_model=512, nhead=8, num_layers=6):
        super().__init__()
        self.d_model = d_model
        self.action_embed = nn.Linear(7, d_model)  # 7-DoF actions
        self.pos_encoder = PositionalEncoding(d_model)

        encoder_layer = nn.TransformerEncoderLayer(
            d_model=d_model, nhead=nhead, batch_first=True)
        self.transformer = nn.TransformerEncoder(
            encoder_layer, num_layers=num_layers)

        self.action_head = nn.Linear(d_model, 7)

    def forward(self, multimodal_features, action_history=None):
        # Embed action history if provided
        if action_history is not None:
            action_emb = self.action_embed(action_history)
            # Concatenate with multimodal features
            features = torch.cat([multimodal_features, action_emb], dim=1)
        else:
            features = multimodal_features

        # Apply positional encoding
        features = self.pos_encoder(features)

        # Process through transformer
        output = self.transformer(features)

        # Generate next action
        next_action = self.action_head(output[:, -1, :])
        return next_action
```

### Memory-Augmented Networks
```python
class MemoryAugmentedVLA(nn.Module):
    def __init__(self):
        super().__init__()
        self.perception = VisionEncoder()
        self.controller = ActionTransformer()
        self.memory = ExternalMemory(
            memory_size=1000, embedding_dim=512)

    def forward(self, observation, instruction, task_embedding=None):
        # Process current observation
        features = self.perception(observation)

        # Retrieve relevant memories
        if task_embedding is not None:
            retrieved = self.memory.query(task_embedding)
            features = torch.cat([features, retrieved], dim=-1)

        # Generate action
        action = self.controller(features)

        # Update memory
        self.memory.write(features, action)

        return action
```
